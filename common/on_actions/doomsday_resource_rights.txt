# Start-date resource rights. HOI4 rights hand every resource in the state to the
# receiver and the owner is not paid, so this list is only extraction the host
# does not sell. Paid offtake, equity mines, prepayments, and production-sharing
# sales are trade and are not here.
# Deterministic. Country-scoped. No random. Flag-guarded so on_startup is once.

on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = dd_resource_rights_init } }
				set_global_flag = dd_resource_rights_init

				# Myanmar ionic-clay rare earths. Informal Kachin and Mandalay
				# shipments into China. The state does not sell this ore.
				BRM = {
					give_resource_rights = { receiver = CHI state = 640 }
				}

				# Liberia rubber. Firestone's concession, not a purchase from Monrovia.
				LIB = {
					give_resource_rights = { receiver = USA state = 298 }
				}
			}
		}
	}
}
