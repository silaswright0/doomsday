on_actions = {
	on_startup = {
		effect = {
			PER = {
				if = {
					limit = {
						OR = {
							has_completed_focus = PER_final_battle
							has_completed_focus = PER_iran_against_the_world
							has_idea = dd_per_final_battle
							has_idea = dd_per_final_battle_2
							has_dynamic_modifier = { modifier = dd_per_final_battle_spirit }
						}
					}
					dd_per_apply_final_battle = yes
				}
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					tag = PER
					NOT = { has_country_flag = dd_per_jan_crackdown }
					date > 2026.1.9
					date < 2026.2.1
				}
				set_country_flag = dd_per_jan_crackdown
				country_event = { id = dd_iran.20 }
			}
			if = {
				limit = { tag = PER }
				dd_per_sync_hezbollah = yes
				dd_irq_close_other_brigades = yes
			}
			if = {
				limit = { has_global_flag = dd_gulf_closed_usa }
				dd_gulf_strip_usa_access = yes
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { SAU = { has_idea = dd_sau_squeezed } }
				SAU = { dd_sau_squeeze_tick = yes }
			}
		}
	}
}
