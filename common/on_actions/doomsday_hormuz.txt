on_actions = {
	on_startup = {
		effect = {
			# Hadramaut is Yemen's. The Council holds the coast and the desert. Seiyun stays with the government.
			if = {
				limit = {
					NOT = { has_global_flag = dd_yem_hadramaut_held }
					659 = { is_owned_by = YEM }
				}
				set_global_flag = dd_yem_hadramaut_held
				STC = {
					set_province_controller = 2029
					set_province_controller = 12821
					set_province_controller = 12894
					set_province_controller = 13079
				}
			}
		}
	}
	on_call_allies = {
		effect = {
			if = {
				limit = { ROOT = { has_idea = dd_usa_iran_war } }
				every_country = {
					limit = {
						has_war_with = FROM
						NOT = { tag = HOU }
						OR = {
							tag = PER
							tag = HAM
							tag = HEZ
							is_subject_of = PER
							has_country_flag = dd_irg_revolt
						}
					}
					white_peace = FROM
				}
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = dd_hormuz_war }
					NOT = { has_global_flag = dd_hormuz_ended }
					date > 2026.2.27
					USA = { exists = yes }
					PER = { exists = yes }
				}
				USA = { country_event = { id = dd_hormuz.1 } }
			}
			if = {
				limit = {
					has_global_flag = dd_hormuz_war
					NOT = { USA = { has_war_with = PER } }
				}
				dd_hormuz_end = yes
			}
			if = {
				limit = {
					tag = USA
					has_global_flag = dd_hormuz_war
				}
				dd_hormuz_sync = yes
				dd_hormuz_auth_daily = yes
				dd_hormuz_econ_sync = yes
				dd_usa_sync_partner_air = yes
			}
		}
	}
	# Every country hits this, and the limit fails immediately unless the war is on and this is the United States.
	on_hourly = {
		effect = {
			if = {
				limit = {
					has_global_flag = dd_hormuz_war
					tag = USA
				}
				dd_hormuz_sync = yes
				dd_usa_sync_partner_air = yes
			}
		}
	}
}
