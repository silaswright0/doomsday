on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = {
						OR = {
							has_completed_focus = ISR_the_idf
							has_completed_focus = ISR_urban_fighting
							has_completed_focus = ISR_combined_arms
							has_idea = dd_isr_idf
							has_idea = dd_isr_idf_urban
							has_idea = dd_isr_idf_arms
							has_idea = dd_isr_idf_both
							has_dynamic_modifier = { modifier = dd_isr_idf_spirit }
						}
						NOT = { has_government = communism }
					}
					dd_isr_refresh_idf = yes
				}
				if = {
					limit = {
						OR = {
							has_completed_focus = ISR_airforce
							has_completed_focus = ISR_air_dominance
							has_idea = dd_isr_airforce
							has_idea = dd_isr_airforce_dominance
							has_dynamic_modifier = { modifier = dd_isr_air_spirit }
						}
					}
					dd_isr_refresh_air = yes
				}
			}
			ISR = {
				if = {
					limit = {
						NOT = { has_country_flag = dd_isr_election_cd_armed }
						NOT = { has_country_flag = dd_isr_election_done }
						date < 2026.1.8
					}
					set_country_flag = dd_isr_election_cd_armed
					activate_mission = dd_isr_election_countdown
				}
				if = {
					limit = { NOT = { has_country_flag = dd_isr_pie_events } }
					set_country_flag = dd_isr_pie_events
					country_event = { id = dd_isr.10 days = 45 }
					country_event = { id = dd_isr.11 days = 140 }
					country_event = { id = dd_isr.12 days = 220 }
				}
				if = {
					limit = { PAL = { exists = yes is_subject_of = ISR } }
					set_country_flag = dd_isr_pal_subject
				}
			}
		}
	}
	on_daily_ISR = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = dd_isr_election_done }
					NOT = { has_country_flag = dd_isr_election_called }
					NOT = { has_country_flag = dd_isr_election_rerun_active }
					date > 2026.10.31
				}
				dd_isr_hold_election = yes
			}
			if = {
				limit = {
					OR = {
						has_government = communism
						has_country_flag = dd_isr_idf_sealed
					}
					OR = {
						has_dynamic_modifier = { modifier = dd_isr_idf_spirit }
						has_idea = dd_isr_idf
						has_idea = dd_isr_idf_urban
						has_idea = dd_isr_idf_arms
						has_idea = dd_isr_idf_both
						has_country_flag = dd_isr_idf_urban
						has_country_flag = dd_isr_idf_arms
					}
				}
				dd_isr_clear_idf = yes
			}
			if = {
				limit = { has_country_flag = dd_isr_diaspora_open }
				dd_isr_diaspora_purge = yes
			}
			if = {
				limit = {
					has_country_flag = dd_isr_pal_subject
					NOT = { has_country_flag = dd_isr_pal_annex_noted }
					NOT = { PAL = { exists = yes } }
				}
				set_country_flag = dd_isr_pal_annex_noted
				country_event = { id = dd_isr.20 }
			}
		}
	}
	on_weekly = {
		effect = {
			every_country = {
				limit = { has_idea = dd_isr_reactionary_ripple }
				add_popularity = { ideology = communism popularity = 0.002 }
			}
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = {
					tag = ISR
					FROM = { tag = PAL }
					NOT = { has_country_flag = dd_isr_pal_annex_noted }
				}
				set_country_flag = dd_isr_pal_annex_noted
				country_event = { id = dd_isr.20 }
			}
		}
	}
}
