# Money, market, and policy. Country-scoped. No random.

on_actions = {
	on_startup = {
		effect = {
			# Hide leftover vanilla NATO decisions (date > 1944). Same PP cost
			# as the 24 Jan crash; complete_effect still fired missing events.
			set_global_flag = brussels_to_fired
			set_global_flag = nato_suggested
			every_country = {
				# Flag-guarded. First country seeds the shared pool; the rest skip.
				dd_init_market_pool = yes
				dd_init_economy = yes
				dd_intel_init = yes
				dd_merc_init_defaults = yes
				dd_sync_csto_spirit = yes
			}
			# Once only. A later load must not put the rule back after it was cleared.
			USA = {
				if = {
					limit = { NOT = { has_country_flag = dd_defensive_pact_set } }
					set_country_flag = dd_defensive_pact_set
					set_faction_rule = dd_call_to_war_defensive_only
				}
			}
			SOV = {
				if = {
					limit = { NOT = { has_country_flag = dd_defensive_pact_set } }
					set_country_flag = dd_defensive_pact_set
					set_faction_rule = dd_call_to_war_defensive_only
				}
			}
		}
	}

	on_monthly = {
		effect = {
			dd_economy_monthly_tick = yes
			dd_intel_cd_tick = yes
			dd_ig_monthly_cleanup = yes
			dd_sync_csto_spirit = yes
		}
	}

	on_yearly = {
		effect = {
			dd_intel_yearly_cost = yes
		}
	}
}
