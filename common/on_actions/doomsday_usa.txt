on_actions = {
	on_startup = {
		effect = {
			USA = {
				if = {
					limit = { NOT = { has_country_flag = dd_usa_calendar_set } }
					set_country_flag = dd_usa_calendar_set
					country_event = { id = dd_usa_election.1 days = 796 }
					country_event = { id = dd_usa_election.2 days = 1041 }
				}
				dd_usa_init_legislature = yes
				dd_usa_grant_station_docking = yes
				dd_usa_mark_starting_states = yes
			}
		}
	}

	on_monthly = {
		effect = {
			dd_usa_israel_arms_shipment = yes
		}
	}

	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					has_idea = dd_usa_provoking_the_blob
					NOT = { has_country_flag = dd_usa_post_election }
				}
				add_to_variable = { dd_usa_blob_months = 1 }
				if = {
					limit = { check_variable = { dd_usa_blob_months > 2 } }
					set_variable = { dd_usa_blob_months = 0 }
					dd_usa_blob_tick = yes
				}
			}
			dd_usa_check_globalists_spirit = yes
		}
	}

	on_daily = {
		effect = {
			dd_usa_cw_strip_diplomacy = yes
		}
	}

	on_daily_USA = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = dd_usa_civil_war_started }
					NOT = { has_country_flag = dd_usa_civil_war }
					has_stability < 0.01
				}
				dd_usa_start_civil_war = yes
			}
			dd_usa_legislature_daily = yes
			dd_usa_me_blood_tick = yes
			dd_usa_check_globalists_spirit = yes
			if = {
				limit = {
					focus_progress = { focus = USA_break_the_republic progress > 0.88 }
					NOT = { has_country_flag = dd_usa_executives_on_trial_news }
				}
				set_country_flag = dd_usa_executives_on_trial_news
				news_event = { id = dd_usa_news.11 }
			}
		}
	}

}
