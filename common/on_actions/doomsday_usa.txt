	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					has_idea = dd_usa_provoking_the_blob
					NOT = { has_country_flag = dd_usa_post_election }
				}
				add_to_variable = { dd_usa_blob_months = 1 }

			}

	}

		effect = {
			# 2% a week, counted out: the 50th week Trump is still leading MAGA.
			if = {
				limit = {
					has_global_flag = dd_usa_civil_war_started

						has_country_leader = { name = "Donald Trump" ruling_only = yes }
						VNM = { has_country_leader = { name = "Donald Trump" ruling_only = yes } }
					}
				}
				add_to_variable = { global.dd_usa_trump_death_weeks = 1 }
				if = {
					limit = { check_variable = { global.dd_usa_trump_death_weeks > 49 } }
					set_global_flag = dd_usa_trump_dead

			}

	}

	}
