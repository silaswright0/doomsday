# Federal Ethiopia is fighting the Tigray rising, or that side has been destroyed.
dd_eth_tigray_open = {
	NOT = { has_country_flag = dd_eth_tig_side }
	OR = {
		any_enemy_country = { has_country_flag = dd_eth_tig_side }
		AND = {
			has_country_flag = dd_eth_rebellion_done
			NOT = {
				any_country = {
					has_country_flag = dd_eth_tig_side
					exists = yes
					has_capitulated = no
				}
			}
		}
	}
}

# The Eritrea war is underway, or Eritrea has been beaten.
dd_eth_eritrea_open = {
	OR = {
		has_war_with = ERI
		ERI = { exists = no }
		ERI = { has_capitulated = yes }
	}
}

# The Sudan war is over: one side is gone, or the two are no longer fighting.
dd_sudan_war_over = {
	OR = {
		RSF = { exists = no }
		SUD = { exists = no }
		NOT = { RSF = { has_war_with = SUD } }
	}
	OR = {
		RSF = { exists = yes has_capitulated = no }
		SUD = { exists = yes has_capitulated = no }
	}
}
