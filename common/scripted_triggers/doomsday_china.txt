# Revolution cells for the Federal Republic and the New Left.
# Country scope unless noted. No random.

dd_chi_rev_unlocked = {
	has_global_flag = dd_chi_civil_war_started
	OR = {
		AND = {
			original_tag = FRC
			has_completed_focus = CHI_federal_republic
		}
		AND = {
			original_tag = NLC
			has_completed_focus = CHI_new_left
		}
	}
}

# State scope. ROOT is the country choosing the state.
# A state locked by Eradicate Resistance in the Cities stays closed.
dd_chi_rev_state_open = {
	NOT = { has_state_flag = dd_chi_city_locked }
	OR = {
		is_core_of = CHI
		is_claimed_by = CHI
	}
	OR = {
		AND = {
			ROOT = { original_tag = FRC }
			NOT = { has_state_flag = dd_frc_rev_prep }
		}
		AND = {
			ROOT = { original_tag = NLC }
			NOT = { has_state_flag = dd_nlc_rev_prep }
		}
	}
}

# State scope. Prepared, the wait has ended, and the cells have not been raised.
dd_chi_rev_state_due = {
	dd_chi_rev_state_ready = yes
	OR = {
		AND = {
			ROOT = { original_tag = FRC }
			NOT = { has_state_flag = dd_frc_rev_wait }
		}
		AND = {
			ROOT = { original_tag = NLC }
			NOT = { has_state_flag = dd_nlc_rev_wait }
		}
	}
}

# State scope. Prepared, and the cells have not been raised. The 60-day wait is separate.
dd_chi_rev_state_ready = {
	NOT = { has_state_flag = dd_chi_city_locked }
	OR = {
		AND = {
			ROOT = { original_tag = FRC }
			has_state_flag = dd_frc_rev_prep
			NOT = { has_state_flag = dd_frc_rev_risen }
		}
		AND = {
			ROOT = { original_tag = NLC }
			has_state_flag = dd_nlc_rev_prep
			NOT = { has_state_flag = dd_nlc_rev_risen }
		}
	}
}

# Character scope. Hired advisors, high command, service chiefs, field marshals, and generals.
# The country leader is left in place.
# is_hired_as_advisor crashes on a character with no advisor role, so generals are tested first
# and the hired check only runs after is_advisor.
dd_chi_purge_eligible = {
	NOT = { is_country_leader = yes }
	OR = {
		is_army_leader = yes
		AND = {
			is_advisor = yes
			is_hired_as_advisor = yes
		}
	}
}

# Country scope. Someone the purge decision can still take.
dd_chi_purge_has_target = {
	any_character = {
		dd_chi_purge_eligible = yes
	}
}

# Liberal democracies. Taiwan's vote and the Pacific concord use this.
dd_chi_is_liberal_democracy = {
	OR = {
		has_government = social_liberalism
		has_government = liberal_conservatism
		has_government = social_democracy
	}
}

# Western capitals for the buyout and the embargo lift.
dd_chi_is_western = {
	OR = {
		original_tag = USA
		original_tag = ENG
		original_tag = FRA
		original_tag = GER
		original_tag = ITA
		original_tag = CAN
		original_tag = AST
		original_tag = HOL
		original_tag = BEL
		original_tag = SPR
		original_tag = POR
	}
}

# All three crises have to still be on the country. The third focus in a column removes that one.
dd_chi_crises_open = {
	has_dynamic_modifier = { modifier = dd_chi_debt_crisis }
	has_dynamic_modifier = { modifier = dd_chi_youth_crisis }
	has_dynamic_modifier = { modifier = dd_chi_estate_crisis }
}

# Country scope, FROM is the state. The wait flag lasts 60 days.
dd_chi_rev_wait_done = {
	OR = {
		AND = {
			original_tag = FRC
			FROM = { NOT = { has_state_flag = dd_frc_rev_wait } }
		}
		AND = {
			original_tag = NLC
			FROM = { NOT = { has_state_flag = dd_nlc_rev_wait } }
		}
	}
}
