# Iranian regime authority. Iran starts at 90% stability, so the war opens on Total Control.
# PER.dd_hormuz_auth_shift moves the card left, toward Total Control, one step per point.
dd_hormuz_auth_1 = {
	OR = {
		PER = { has_stability > 0.79 }
		PER = {
			check_variable = { dd_hormuz_auth_shift > 0 }
			has_stability > 0.59
		}
		PER = {
			check_variable = { dd_hormuz_auth_shift > 1 }
			has_stability > 0.39
		}
		PER = {
			check_variable = { dd_hormuz_auth_shift > 2 }
			has_stability > 0.19
		}
		PER = { check_variable = { dd_hormuz_auth_shift > 3 } }
	}
}
dd_hormuz_auth_2 = {
	OR = {
		PER = {
			NOT = { check_variable = { dd_hormuz_auth_shift > 0 } }
			has_stability > 0.59
		}
		PER = {
			check_variable = { dd_hormuz_auth_shift = 1 }
			has_stability > 0.39
			NOT = { has_stability > 0.59 }
		}
		PER = {
			check_variable = { dd_hormuz_auth_shift = 2 }
			has_stability > 0.19
			NOT = { has_stability > 0.39 }
		}
		PER = {
			check_variable = { dd_hormuz_auth_shift = 3 }
			NOT = { has_stability > 0.19 }
		}
	}
	NOT = { dd_hormuz_auth_1 = yes }
}
