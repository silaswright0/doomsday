dd_usa_cancel_elections = {
	if = {
		limit = { has_government = fascism }
		set_politics = { ruling_party = fascism elections_allowed = no }
	}
	else_if = {
		limit = { has_government = communism }
		set_politics = { ruling_party = communism elections_allowed = no }
	}
	else_if = {
		limit = { has_government = anarcho_communism }
		set_politics = { ruling_party = anarcho_communism elections_allowed = no }
	}
	else_if = {
		limit = { has_government = democratic_confederalism }
		set_politics = { ruling_party = democratic_confederalism elections_allowed = no }
	}
	else = {
		set_politics = { ruling_party = national_populism elections_allowed = no }
	}
}

dd_usa_cut_state_pop_percent = {
	set_temp_variable = { dd_cut = state_population_k }
	multiply_temp_variable = { dd_cut = 10 }
	round_temp_variable = dd_cut
	multiply_temp_variable = { dd_cut = -1 }
	set_variable = { dd_pop_cut = dd_cut }
	if = {
		limit = { check_variable = { dd_pop_cut < 0 } }
		add_manpower = var:dd_pop_cut
	}
	clear_variable = dd_pop_cut
}

dd_usa_abaddon_strike = {
	every_state = {
		save_event_target_as = dd_abaddon_state
		PREV = {
			if = {
				limit = { num_of_nukes > 0 }
				launch_nuke = {
					state = event_target:dd_abaddon_state
					use_nuke = yes
				}
			}
		}
		damage_building = { type = infrastructure damage = 5 }
		add_manpower = -500000
	}
	news_event = { id = dd_usa_abaddon.2 }
	country_event = { id = dd_usa_abaddon.1 days = 7 }
}

dd_usa_abaddon_end = {
	every_country = {
		every_enemy_country = {
			white_peace = PREV
		}
	}
	every_state = {
		limit = { NOT = { is_owned_by = ABY } }
		ABY = { transfer_state = PREV }
	}
	every_country = {
		limit = { NOT = { tag = ABY } }
		add_manpower = -10000000
	}
	ABY = {
		set_cosmetic_tag = ABY_WASTE
		add_manpower = -10000000
	}
}

dd_usa_refresh_invite_targets = {
	clear_array = global.dd_left_invite
	every_country = {
		limit = {
			exists = yes
			OR = {
				has_government = progressive_populism
				has_government = anarcho_communism
				has_government = communism
				has_government = social_democracy
				has_government = left_wing_nationalism
				has_government = democratic_confederalism
			}
		}
		add_to_array = { global.dd_left_invite = THIS }
	}
}
