# France takes the chair, then the United States is out. Subjects are pulled out first.
# dismantle_faction releases the leader's puppets, and a subject left inside NATO is freed
# the moment Washington goes to war with that alliance.
dd_usa_hand_nato_to_france = {
	if = {
		limit = { is_in_faction = yes }
		if = {
			limit = { is_faction_leader = yes }
			every_country = {
				limit = { is_subject_of = ROOT }
				if = {
					limit = { has_autonomy_state = autonomy_integrated_puppet }
					add_to_array = { ROOT.dd_sub_integrated = THIS }
				}
				else_if = {
					limit = { has_autonomy_state = autonomy_dominion }
					add_to_array = { ROOT.dd_sub_dominion = THIS }
				}
				else = {
					add_to_array = { ROOT.dd_sub_puppet = THIS }
				}
				if = {
					limit = { is_in_faction = yes }
					remove_from_faction = yes
				}
			}
			every_other_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { is_subject_of = ROOT }
				}
				add_to_array = { ROOT.dd_nato_remain = THIS }
			}
			dismantle_faction = yes
			FRA = {
				create_faction = NATO
				set_faction_rule = dd_call_to_war_defensive_only
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_nato_remain value = THIS }
					NOT = { tag = FRA }
					NOT = { tag = USA }
					NOT = { is_subject_of = ROOT }
				}
				FRA = { add_to_faction = PREV }
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_integrated value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_integrated_puppet
						freedom_level = 0.2
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_dominion value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_dominion
						freedom_level = 0.85
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_puppet value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_puppet
						freedom_level = 0.4
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			clear_array = ROOT.dd_nato_remain
			clear_array = ROOT.dd_sub_integrated
			clear_array = ROOT.dd_sub_dominion
			clear_array = ROOT.dd_sub_puppet
		}
		else = {
			remove_from_faction = yes
		}
	}
}

# 100 political power, or 10 when any other board country is ahead on trade war points.
dd_tw_refresh_costs = {
	set_variable = { dd_tw_decision_cost = 100 }
	if = {
		limit = {
			NOT = { tag = USA }
			USA = { exists = yes has_variable = dd_trade_war_points }
			check_variable = { dd_trade_war_points < USA.dd_trade_war_points }
		}
		set_variable = { dd_tw_decision_cost = 10 }
	}
	if = {
		limit = {
			NOT = { tag = CAN }
			CAN = { exists = yes has_variable = dd_trade_war_points }
			check_variable = { dd_trade_war_points < CAN.dd_trade_war_points }
		}
		set_variable = { dd_tw_decision_cost = 10 }
	}
	if = {
		limit = {
			NOT = { tag = MEX }
			MEX = { exists = yes has_variable = dd_trade_war_points }
			check_variable = { dd_trade_war_points < MEX.dd_trade_war_points }
		}
		set_variable = { dd_tw_decision_cost = 10 }
	}
}

# Tariff board. Caller sets temp dd_tw_add_cg and dd_tw_add_fac, then notes which partner the hit belongs to.
dd_tw_ensure_spirit = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = dd_tariff_war } } }
		if = { limit = { NOT = { has_variable = dd_tariff_cg } } set_variable = { dd_tariff_cg = 0 } }
		if = { limit = { NOT = { has_variable = dd_tariff_fac } } set_variable = { dd_tariff_fac = 0 } }
		if = { limit = { NOT = { has_variable = dd_trade_war_points } } set_variable = { dd_trade_war_points = 0 } }
		add_dynamic_modifier = { modifier = dd_tariff_war }
	}
	add_to_variable = { dd_tariff_cg = dd_tw_add_cg }
	add_to_variable = { dd_tariff_fac = dd_tw_add_fac }
}

dd_tw_clamp_spirit = {
	if = { limit = { NOT = { has_variable = dd_tariff_cg } } set_variable = { dd_tariff_cg = 0 } }
	if = { limit = { check_variable = { dd_tariff_cg < 0.0001 } } set_variable = { dd_tariff_cg = 0 } }
	if = { limit = { NOT = { has_variable = dd_tariff_fac } } set_variable = { dd_tariff_fac = 0 } }
	if = { limit = { check_variable = { dd_tariff_fac > -0.0001 } } set_variable = { dd_tariff_fac = 0 } }
	if = { limit = { NOT = { has_variable = dd_trade_war_points } } set_variable = { dd_trade_war_points = 0 } }
	if = { limit = { check_variable = { dd_trade_war_points < 0.0001 } } set_variable = { dd_trade_war_points = 0 } }
	if = {
		limit = {
			has_dynamic_modifier = { modifier = dd_tariff_war }
			check_variable = { dd_tariff_cg = 0 }
			check_variable = { dd_tariff_fac = 0 }
		}
		remove_dynamic_modifier = { modifier = dd_tariff_war }
	}
	dd_tw_refresh_all_costs = yes
}

# Day the measure is taken, plus which measure. Even sum: the neighbor answers
# with one of its own tariffs. Same day and same measure always agree in MP.
dd_tw_ai_answer_can = {
	if = {
		limit = {
			CAN = { exists = yes is_ai = yes }
			NOT = { has_country_flag = dd_tw_settled_can }
		}
		set_temp_variable = { dd_tw_flip = global.num_days }
		add_to_temp_variable = { dd_tw_flip = dd_tw_measure }
		modulo_temp_variable = { dd_tw_flip = 2 }
		if = {
			limit = { check_variable = { dd_tw_flip = 0 } }
			CAN = { dd_tw_fire_next_can = yes }
		}
	}
}

dd_tw_ai_answer_mex = {
	if = {

			MEX = { exists = yes is_ai = yes }
			NOT = { has_country_flag = dd_tw_settled_mex }
		}

		if = {
			limit = { check_variable = { dd_tw_flip = 0 } }

		}
	}
}

	remove_ideas = dd_usa_civil_war_cause
	if = {
		limit = { NOT = { has_country_flag = dd_usa_civil_war_over } }
		clr_country_flag = dd_usa_civil_war
		set_country_flag = dd_usa_civil_war_over
		dd_usa_apply_nation_in_ruins = yes
		if = {
			limit = { NOT = { has_idea = dd_usa_space_infrastructure } }
			add_ideas = dd_usa_space_infrastructure
		}
		country_event = { id = dd_usa_recon.1 }
	}
}

dd_usa_apply_nation_in_ruins = {
	if = { limit = { NOT = { has_variable = dd_ruins_stability } } set_variable = { dd_ruins_stability = -0.15 } }
	if = { limit = { NOT = { has_variable = dd_ruins_war_support } } set_variable = { dd_ruins_war_support = -0.25 } }
	if = { limit = { NOT = { has_variable = dd_ruins_factory } } set_variable = { dd_ruins_factory = -0.25 } }
	if = { limit = { NOT = { has_variable = dd_ruins_cg } } set_variable = { dd_ruins_cg = 0.30 } }
	if = {
		limit = { has_dynamic_modifier = { modifier = dd_usa_nation_in_ruins } }
		remove_dynamic_modifier = { modifier = dd_usa_nation_in_ruins }
	}
	add_dynamic_modifier = { modifier = dd_usa_nation_in_ruins }
	force_update_dynamic_modifier = yes
	country_event = { id = dd_spirit.1 days = 1 }
}

# Little Tech swaps one idea card per tier. The level is set by the focus before this runs.
# A loaded save with an older tier, or the empty variable spirit, is moved onto the matching card.
dd_usa_apply_little_tech = {
	if = {
		limit = { NOT = { check_variable = { dd_usa_little_tech_level > 0 } } }
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech
					has_completed_focus = USA_little_tech
				}
			}
			set_variable = { dd_usa_little_tech_level = 1 }
		}
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech_2
					has_completed_focus = USA_bring_in_little_tech
				}
			}
			set_variable = { dd_usa_little_tech_level = 2 }
		}
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech_3
					has_completed_focus = USA_prepare_for_agi
				}
			}
			set_variable = { dd_usa_little_tech_level = 3 }
		}
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech_4
					has_completed_focus = USA_final_stretch
				}
			}
			set_variable = { dd_usa_little_tech_level = 4 }
		}
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech_5
					has_completed_focus = USA_ai_administration
				}
			}
			set_variable = { dd_usa_little_tech_level = 5 }
		}
		if = {
			limit = {
				OR = {
					has_idea = dd_usa_little_tech_6
					has_completed_focus = USA_capitalisms_final_form
				}
			}
			set_variable = { dd_usa_little_tech_level = 6 }
		}
	}
	if = {
		limit = { check_variable = { dd_usa_little_tech_level > 0 } }
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_usa_little_tech_spirit } }
			remove_dynamic_modifier = { modifier = dd_usa_little_tech_spirit }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level = 1 } } }
			if = { limit = { has_idea = dd_usa_little_tech } remove_ideas = dd_usa_little_tech }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level = 2 } } }
			if = { limit = { has_idea = dd_usa_little_tech_2 } remove_ideas = dd_usa_little_tech_2 }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level = 3 } } }
			if = { limit = { has_idea = dd_usa_little_tech_3 } remove_ideas = dd_usa_little_tech_3 }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level = 4 } } }
			if = { limit = { has_idea = dd_usa_little_tech_4 } remove_ideas = dd_usa_little_tech_4 }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level = 5 } } }
			if = { limit = { has_idea = dd_usa_little_tech_5 } remove_ideas = dd_usa_little_tech_5 }
		}
		if = {
			limit = { NOT = { check_variable = { dd_usa_little_tech_level > 5 } } }
			if = { limit = { has_idea = dd_usa_little_tech_6 } remove_ideas = dd_usa_little_tech_6 }
		}
		if = {
			limit = { check_variable = { dd_usa_little_tech_level = 1 } }
			if = { limit = { NOT = { has_idea = dd_usa_little_tech } } add_ideas = dd_usa_little_tech }
		}
		else_if = {
			limit = { check_variable = { dd_usa_little_tech_level = 2 } }
			if = { limit = { NOT = { has_idea = dd_usa_little_tech_2 } } add_ideas = dd_usa_little_tech_2 }
		}
		else_if = {
			limit = { check_variable = { dd_usa_little_tech_level = 3 } }
			if = { limit = { NOT = { has_idea = dd_usa_little_tech_3 } } add_ideas = dd_usa_little_tech_3 }
		}
		else_if = {
			limit = { check_variable = { dd_usa_little_tech_level = 4 } }
			if = { limit = { NOT = { has_idea = dd_usa_little_tech_4 } } add_ideas = dd_usa_little_tech_4 }
		}
		else_if = {
			limit = { check_variable = { dd_usa_little_tech_level = 5 } }
			if = { limit = { NOT = { has_idea = dd_usa_little_tech_5 } } add_ideas = dd_usa_little_tech_5 }
		}
		else = {
			if = { limit = { NOT = { has_idea = dd_usa_little_tech_6 } } add_ideas = dd_usa_little_tech_6 }
		}
	}
}

# Gutting the Middle Class is an idea. Dismantle White America swaps to the cleared card.
dd_usa_apply_gut = {
	if = {
		limit = {
			OR = {
				has_idea = dd_usa_gutting_middle_class_2
				has_completed_focus = USA_dismantle_white_america
			}
		}
		set_country_flag = dd_usa_gut_cleared
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = dd_usa_gut_spirit } }
		remove_dynamic_modifier = { modifier = dd_usa_gut_spirit }
	}
	if = {
		limit = { has_country_flag = dd_usa_gut_cleared }
		if = { limit = { has_idea = dd_usa_gutting_middle_class } remove_ideas = dd_usa_gutting_middle_class }
		if = { limit = { NOT = { has_idea = dd_usa_gutting_middle_class_2 } } add_ideas = dd_usa_gutting_middle_class_2 }
	}
	else = {
		if = { limit = { has_idea = dd_usa_gutting_middle_class_2 } remove_ideas = dd_usa_gutting_middle_class_2 }
		if = { limit = { NOT = { has_idea = dd_usa_gutting_middle_class } } add_ideas = dd_usa_gutting_middle_class }
	}
}

# The United States, those states, and the rest of NATO leave the war.
# Israel is put back in if the faction peace dragged it out.

	dd_usa_surrender_persian_gulf = yes
	if = {
		limit = { ISR = { has_war_with = PER } }
		set_country_flag = dd_isr_keeps_iran_war
	}
	USA = { set_country_flag = dd_iran_win_exit }
	BHR = { set_country_flag = dd_iran_win_exit }
	IRQ = { set_country_flag = dd_iran_win_exit }
	KUW = { set_country_flag = dd_iran_win_exit }
	QAT = { set_country_flag = dd_iran_win_exit }
	SAU = { set_country_flag = dd_iran_win_exit }
	UAE = { set_country_flag = dd_iran_win_exit }
	OMA = { set_country_flag = dd_iran_win_exit }
	every_country = {
		limit = {
			exists = yes
			is_in_faction_with = USA
			NOT = { tag = ISR }
		}
		set_country_flag = dd_iran_win_exit
	}
	if = {
		limit = { has_idea = dd_usa_iran_war }
		remove_ideas = dd_usa_iran_war
	}
	every_country = {
		limit = {
			has_country_flag = dd_iran_win_exit
			has_war_with = PER
		}
		white_peace = PER
	}
	PER = {
		every_enemy_country = {
			limit = { has_country_flag = dd_iran_win_exit }
			white_peace = PREV
		}
	}
	every_country = {
		limit = {
			OR = {
				tag = HAM
				tag = HEZ
				tag = HOU
				is_subject_of = PER
			}
		}
		every_enemy_country = {
			limit = { has_country_flag = dd_iran_win_exit }
			white_peace = PREV
		}
	}
	every_country = {
		limit = { has_country_flag = dd_iran_win_exit }
		clr_country_flag = dd_iran_win_exit
	}
	if = {
		limit = {
			has_country_flag = dd_isr_keeps_iran_war
			NOT = { ISR = { has_war_with = PER } }
		}
		ISR = { declare_war_on = { target = PER type = annex_everything call_allies = no } }
	}
	clr_country_flag = dd_isr_keeps_iran_war
}

dd_usa_decay_lobby_dem = {
	set_temp_variable = { dd_usa_lobby_i = 0 }
	while = {
		limit = { check_variable = { dd_usa_lobby_i < dd_usa_lobby_dem_days^num } }
		subtract_from_variable = { dd_usa_lobby_dem_days^dd_usa_lobby_i = 1 }
		if = {
			limit = { check_variable = { dd_usa_lobby_dem_days^dd_usa_lobby_i < 1 } }
			remove_from_array = { dd_usa_lobby_dem_days = dd_usa_lobby_i }
		}
		else = {
			add_to_temp_variable = { dd_usa_lobby_i = 1 }
		}
	}
}

dd_usa_refresh_lobby_seats = {
	set_temp_variable = { dd_usa_lobby_net = dd_usa_lobby_dem_days^num }
	subtract_from_temp_variable = { dd_usa_lobby_net = dd_usa_lobby_rep_days^num }
	set_temp_variable = { dd_usa_lobby_shift = dd_usa_lobby_net }
	multiply_temp_variable = { dd_usa_lobby_shift = 6 }
	set_variable = { dd_usa_house_dem_now = dd_usa_house_dem }
	add_to_variable = { dd_usa_house_dem_now = dd_usa_lobby_shift }
	set_temp_variable = { dd_usa_lobby_total = dd_usa_house_dem }
	add_to_temp_variable = { dd_usa_lobby_total = dd_usa_house_rep }
	if = {
		limit = { check_variable = { dd_usa_house_dem_now < 0 } }
		set_variable = { dd_usa_house_dem_now = 0 }
	}
	if = {
		limit = { check_variable = { dd_usa_house_dem_now > dd_usa_lobby_total } }
		set_variable = { dd_usa_house_dem_now = dd_usa_lobby_total }
	}
	set_variable = { dd_usa_house_rep_now = dd_usa_lobby_total }
	subtract_from_variable = { dd_usa_house_rep_now = dd_usa_house_dem_now }
	set_temp_variable = { dd_usa_lobby_shift = dd_usa_lobby_net }
	multiply_temp_variable = { dd_usa_lobby_shift = 2 }
	set_variable = { dd_usa_senate_dem_now = dd_usa_senate_dem }
	add_to_variable = { dd_usa_senate_dem_now = dd_usa_lobby_shift }
	set_temp_variable = { dd_usa_lobby_total = dd_usa_senate_dem }
	add_to_temp_variable = { dd_usa_lobby_total = dd_usa_senate_rep }
	if = {
		limit = { check_variable = { dd_usa_senate_dem_now < 0 } }
		set_variable = { dd_usa_senate_dem_now = 0 }
	}
	if = {
		limit = { check_variable = { dd_usa_senate_dem_now > dd_usa_lobby_total } }
		set_variable = { dd_usa_senate_dem_now = dd_usa_lobby_total }
	}
	set_variable = { dd_usa_senate_rep_now = dd_usa_lobby_total }
	subtract_from_variable = { dd_usa_senate_rep_now = dd_usa_senate_dem_now }
}

			create_country_leader = { name = "Marco Rubio" picture = GFX_portrait_RBM_leader ideology = liberal_conservative }
		}
		if = {
			limit = { check_variable = { dd_usa_general_winner = 4 } }
			set_politics = { ruling_party = national_populism elections_allowed = no }
			create_country_leader = { name = "JD Vance" picture = GFX_portrait_VNM_leader ideology = national_populist }
		}
		dd_usa_sync_caucus_from_ruling = yes
	}
	if = {
		limit = {
			has_completed_focus = USA_path_alt_right
			NOT = { has_country_flag = dd_usa_alt_offered }
			NOT = { has_country_flag = dd_usa_fascists_denied }
		}
		set_country_flag = dd_usa_alt_offered
		country_event = { id = dd_usa_alt.1 days = 7 }
	}
}

dd_usa_run_general = {
	if = {
		limit = { NOT = { has_country_flag = dd_usa_general_resolved } }
		set_country_flag = dd_usa_general_resolved
		if = {
			limit = { check_variable = { dd_usa_left_nominee < 1 } }
			set_variable = { dd_usa_left_nominee = 1 }
		}
		if = {
			limit = { check_variable = { dd_usa_right_nominee < 1 } }
			set_variable = { dd_usa_right_nominee = 1 }
		}
		set_temp_variable = { dd_left = party_popularity@social_democracy }
		add_to_temp_variable = { dd_left = party_popularity@progressive_populism }
		add_to_temp_variable = { dd_left = party_popularity@communism }
		add_to_temp_variable = { dd_left = party_popularity@anarcho_communism }
		add_to_temp_variable = { dd_left = party_popularity@social_liberalism }
		if = {
			limit = { check_variable = { dd_usa_right_nominee = 3 } }
			add_to_temp_variable = { dd_left = party_popularity@liberal_conservatism }
			set_temp_variable = { dd_vance = party_popularity@national_populism }
			set_temp_variable = { dd_fas = party_popularity@fascism }
			set_variable = { dd_usa_general_winner = 5 }
			if = {
				limit = {
					NOT = { check_variable = { dd_fas > dd_left } }
				}
				if = {
					limit = { check_variable = { dd_vance > dd_left } }
					set_variable = { dd_usa_general_winner = 4 }
				}
				else = {
					set_variable = { dd_usa_general_winner = dd_usa_left_nominee }
				}
			}
			if = {
				limit = {
					check_variable = { dd_fas > dd_left }
					NOT = { check_variable = { dd_fas > dd_vance } }
				}
				if = {
					limit = { check_variable = { dd_vance > dd_left } }
					set_variable = { dd_usa_general_winner = 4 }
				}
				else = {
					set_variable = { dd_usa_general_winner = dd_usa_left_nominee }
				}
			}
		}
		else = {
			set_temp_variable = { dd_right = party_popularity@liberal_conservatism }
			add_to_temp_variable = { dd_right = party_popularity@national_populism }
			if = {
				limit = { check_variable = { dd_right > dd_left } }
				set_variable = { dd_usa_general_winner = 3 }
				if = {
					limit = { check_variable = { dd_usa_right_nominee = 2 } }
					set_variable = { dd_usa_general_winner = 4 }
				}
			}
			else = {
				set_variable = { dd_usa_general_winner = dd_usa_left_nominee }
			}
		}
		if = {
			limit = { check_variable = { dd_usa_general_winner = 5 } }
			set_country_flag = dd_usa_fascists_denied
			set_country_flag = dd_usa_post_election
			country_event = { id = dd_usa_election.4 }
		}
		else = {
			dd_usa_seat_president = yes
			news_event = { id = dd_usa_news.15 }
		}
	}
}

dd_usa_election_calendar = {
	if = {
		limit = {
			NOT = { has_global_flag = dd_usa_civil_war_started }
			NOT = { has_country_flag = dd_usa_civil_war }
			NOT = { has_country_flag = dd_usa_elections_closed }
		}
		# The 2026 midterms, the 2028 primaries, and the 2028 general also have
		# mission countdowns, armed on a new game. Later midterms are not played.
		if = {
			limit = {
				date > 2026.11.2
				NOT = { has_country_flag = dd_usa_midterm_2026 }
			}
			set_country_flag = dd_usa_midterm_2026
			country_event = { id = dd_usa_midterm.1 }
		}
		if = {
			limit = {
				date > 2028.3.6
				NOT = { has_country_flag = dd_usa_primary_2028 }
			}
			set_country_flag = dd_usa_primary_2028
			dd_usa_run_primaries = yes
		}
		if = {
			limit = {
				date > 2028.11.6
				NOT = { has_country_flag = dd_usa_general_2028 }
			}
			set_country_flag = dd_usa_general_2028
			dd_usa_run_general = yes
		}
		if = {
			limit = {
				date > 2032.3.6
				NOT = { has_country_flag = dd_usa_primary_2032 }
			}
			set_country_flag = dd_usa_primary_2032
			clr_country_flag = dd_usa_primaries_resolved
			dd_usa_run_primaries = yes
		}
		if = {
			limit = {
				date > 2032.11.6
				NOT = { has_country_flag = dd_usa_general_2032 }
			}
			set_country_flag = dd_usa_general_2032
			clr_country_flag = dd_usa_general_resolved
			dd_usa_run_general = yes
		}
		if = {
			limit = {
				date > 2036.3.6
				NOT = { has_country_flag = dd_usa_primary_2036 }
			}
			set_country_flag = dd_usa_primary_2036
			clr_country_flag = dd_usa_primaries_resolved
			dd_usa_run_primaries = yes
		}
		if = {
			limit = {
				date > 2036.11.6
				NOT = { has_country_flag = dd_usa_general_2036 }
			}
			set_country_flag = dd_usa_general_2036
			clr_country_flag = dd_usa_general_resolved
			dd_usa_run_general = yes
		}
	}
}
