# France takes the chair, then the United States is out. Subjects are pulled out first.
# dismantle_faction releases the leader's puppets, and a subject left inside NATO is freed
# the moment Washington goes to war with that alliance.
dd_usa_hand_nato_to_france = {
	if = {
		limit = { is_in_faction = yes }
		if = {
			limit = { is_faction_leader = yes }
			every_country = {
				limit = { is_subject_of = ROOT }
				if = {
					limit = { has_autonomy_state = autonomy_integrated_puppet }
					add_to_array = { ROOT.dd_sub_integrated = THIS }
				}
				else_if = {
					limit = { has_autonomy_state = autonomy_dominion }
					add_to_array = { ROOT.dd_sub_dominion = THIS }
				}
				else = {
					add_to_array = { ROOT.dd_sub_puppet = THIS }
				}
				if = {
					limit = { is_in_faction = yes }
					remove_from_faction = yes
				}
			}
			every_other_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { is_subject_of = ROOT }
				}
				add_to_array = { ROOT.dd_nato_remain = THIS }
			}
			dismantle_faction = yes
			FRA = {
				create_faction = NATO
				set_faction_rule = dd_call_to_war_defensive_only
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_nato_remain value = THIS }
					NOT = { tag = FRA }
					NOT = { tag = USA }
					NOT = { is_subject_of = ROOT }
				}
				FRA = { add_to_faction = PREV }
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_integrated value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_integrated_puppet
						freedom_level = 0.2
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_dominion value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_dominion
						freedom_level = 0.85
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_puppet value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_puppet
						freedom_level = 0.4
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			clear_array = ROOT.dd_nato_remain
			clear_array = ROOT.dd_sub_integrated
			clear_array = ROOT.dd_sub_dominion
			clear_array = ROOT.dd_sub_puppet
		}
		else = {
			remove_from_faction = yes
		}
	}
}
