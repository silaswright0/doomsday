# France takes the chair, then the United States is out. Subjects are pulled out first.
# dismantle_faction releases the leader's puppets, and a subject left inside NATO is freed
# the moment Washington goes to war with that alliance.
dd_usa_hand_nato_to_france = {
	if = {
		limit = { is_in_faction = yes }
		if = {
			limit = { is_faction_leader = yes }
			every_country = {
				limit = { is_subject_of = ROOT }
				if = {
					limit = { has_autonomy_state = autonomy_integrated_puppet }
					add_to_array = { ROOT.dd_sub_integrated = THIS }
				}
				else_if = {
					limit = { has_autonomy_state = autonomy_dominion }
					add_to_array = { ROOT.dd_sub_dominion = THIS }
				}
				else = {
					add_to_array = { ROOT.dd_sub_puppet = THIS }
				}
				if = {
					limit = { is_in_faction = yes }
					remove_from_faction = yes
				}
			}
			every_other_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { is_subject_of = ROOT }
				}
				add_to_array = { ROOT.dd_nato_remain = THIS }
			}
			dismantle_faction = yes
			FRA = {
				create_faction = NATO
				set_faction_rule = dd_call_to_war_defensive_only
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_nato_remain value = THIS }
					NOT = { tag = FRA }
					NOT = { tag = USA }
					NOT = { is_subject_of = ROOT }
				}
				FRA = { add_to_faction = PREV }
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_integrated value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_integrated_puppet
						freedom_level = 0.2
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_dominion value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_dominion
						freedom_level = 0.85
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			every_country = {
				limit = {
					is_in_array = { array = ROOT.dd_sub_puppet value = THIS }
					is_subject = no
				}
				ROOT = {
					set_autonomy = {
						target = PREV
						autonomous_state = autonomy_puppet
						freedom_level = 0.4
					}
				}
				dd_usa_restore_known_puppet = yes
			}
			clear_array = ROOT.dd_nato_remain
			clear_array = ROOT.dd_sub_integrated
			clear_array = ROOT.dd_sub_dominion
			clear_array = ROOT.dd_sub_puppet
		}
		else = {
			remove_from_faction = yes
		}
	}
}

dd_usa_cw_spawn_militia = {
	dd_usa_cw_ensure_militia_template = yes
	set_variable = { dd_usa_militia_count = 6 }
	if = {
		limit = { has_government = anarcho_communism }
		set_variable = { dd_usa_militia_count = 12 }
	}
	else_if = {
		limit = { has_government = communism }
		set_variable = { dd_usa_militia_count = 10 }
	}
	else_if = {
		limit = { has_government = fascism }
		set_variable = { dd_usa_militia_count = 8 }
	}
	add_equipment_to_stockpile = { type = infantry_equipment_cw1 amount = 6000 }
	# Spread across owned states (same pattern as CW war spawns).
	while_loop_effect = {
		limit = { check_variable = { dd_usa_militia_count > 0 } }
		set_temp_variable = { dd_cw_pass = 0 }
		every_owned_state = {
			limit = {
				is_owned_and_controlled_by = PREV
				check_variable = { PREV.dd_usa_militia_count > 0 }
			}
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
				owner = PREV
				count = 1
			}
			PREV = { subtract_from_variable = { dd_usa_militia_count = 1 } }
			add_to_temp_variable = { dd_cw_pass = 1 }
		}
		if = {
			limit = { check_variable = { dd_cw_pass < 1 } }
			capital_scope = {
				create_unit = {
					division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
					owner = PREV
					count = 1
				}
			}
			subtract_from_variable = { dd_usa_militia_count = 1 }
		}
	}
	clear_variable = dd_usa_militia_count
}

# Daily: caucus support (50% normal / 60% under filibuster) + chamber majority.
dd_usa_legislature_daily = {
	set_variable = { dd_usa_bill_advancing = 0 }
	if = {
		limit = {
			check_variable = { dd_usa_bill_active > 0 }
			NOT = { has_country_flag = dd_usa_civil_war }
		}
		if = {
			limit = { check_variable = { dd_usa_bill_party < 1 } }
			set_temp_variable = { dd_usa_tmp_support = dd_usa_dem_support }
			set_temp_variable = { dd_usa_tmp_house = dd_usa_house_dem }
			set_temp_variable = { dd_usa_tmp_senate = dd_usa_senate_dem }
		}
		else = {
			set_temp_variable = { dd_usa_tmp_support = dd_usa_rep_support }
			set_temp_variable = { dd_usa_tmp_house = dd_usa_house_rep }
			set_temp_variable = { dd_usa_tmp_senate = dd_usa_senate_rep }
		}
		# House and filibuster-cleared Senate: 50%. Active filibuster: 60%.
		set_temp_variable = { dd_usa_need = 50 }
		if = {
			limit = {
				check_variable = { dd_usa_bill_chamber = 1 }
				check_variable = { dd_usa_bill_filibuster_cleared < 1 }
				NOT = { check_variable = { dd_usa_tmp_senate > 59.999 } }
			}
			set_temp_variable = { dd_usa_need = 60 }
		}
		set_variable = { dd_usa_bill_support_need = dd_usa_need }
		if = {
			limit = { check_variable = { dd_usa_tmp_support < dd_usa_need } }
		}
		else_if = {
			limit = {
				check_variable = { dd_usa_bill_chamber < 1 }
				check_variable = { dd_usa_tmp_house > 217.999 }
			}
			set_variable = { dd_usa_bill_advancing = 1 }
			subtract_from_variable = { dd_usa_bill_days_left = 1 }
			if = {
				limit = { check_variable = { dd_usa_bill_days_left < 1 } }
				dd_usa_advance_bill_chamber = yes
			}
		}
		else_if = {
			limit = {
				check_variable = { dd_usa_bill_chamber = 1 }
				check_variable = { dd_usa_tmp_senate > 50.999 }
				OR = {
					check_variable = { dd_usa_tmp_senate > 59.999 }
					check_variable = { dd_usa_bill_filibuster_cleared > 0 }
					check_variable = { dd_usa_tmp_support > 59.999 }
				}
			}
			set_variable = { dd_usa_bill_advancing = 1 }
			subtract_from_variable = { dd_usa_bill_days_left = 1 }
			if = {
				limit = { check_variable = { dd_usa_bill_days_left < 1 } }
				dd_usa_advance_bill_chamber = yes
			}
		}
	}
}
