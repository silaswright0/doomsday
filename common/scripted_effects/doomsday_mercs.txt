# Mercenary market. Country-scoped. Deterministic. No random.
# Seller offers 1 division of a chosen template (volunteer-style pick).
# Buyer may rent only 1 merc division at a time.

# Template ids (shared catalog — matches locked 2026 OOBs):
# 1 Infantry Division | 2 Armored Division | 3 Mechanized Brigade
# 4 Marine Division | 5 Mountain Division | 6 Garrison Brigade | 7 Militia Brigade
# Deployed merc units use create_unit id = 77001 for recall.

dd_merc_refresh_templates = {
	clear_array = dd_merc_tpls
	if = {
		limit = { has_template = "Infantry Division" }
		add_to_array = { dd_merc_tpls = 1 }
	}
	if = {
		limit = { has_template = "Armored Division" }
		add_to_array = { dd_merc_tpls = 2 }
	}
	if = {
		limit = { has_template = "Mechanized Brigade" }
		add_to_array = { dd_merc_tpls = 3 }
	}
	if = {
		limit = { has_template = "Marine Division" }
		add_to_array = { dd_merc_tpls = 4 }
	}
	if = {
		limit = { has_template = "Mountain Division" }
		add_to_array = { dd_merc_tpls = 5 }
	}
	if = {
		limit = { has_template = "Garrison Brigade" }
		add_to_array = { dd_merc_tpls = 6 }
	}
	if = {
		limit = { has_template = "Militia Brigade" }
		add_to_array = { dd_merc_tpls = 7 }
	}
	# Keep prior selection if still available; else first entry.
	if = {
		limit = { check_variable = { dd_merc_tpls^num > 0 } }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = dd_merc_tpl }
					check_variable = { dd_merc_tpl < 1 }
					check_variable = { dd_merc_tpl > 7 }
					AND = {
						check_variable = { dd_merc_tpl = 1 }
						NOT = { has_template = "Infantry Division" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 2 }
						NOT = { has_template = "Armored Division" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 3 }
						NOT = { has_template = "Mechanized Brigade" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 4 }
						NOT = { has_template = "Marine Division" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 5 }
						NOT = { has_template = "Mountain Division" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 6 }
						NOT = { has_template = "Garrison Brigade" }
					}
					AND = {
						check_variable = { dd_merc_tpl = 7 }
						NOT = { has_template = "Militia Brigade" }
					}
				}
			}
			set_variable = { dd_merc_tpl = dd_merc_tpls^0 }
		}
	}
	else = {
		clear_variable = dd_merc_tpl
	}
}

dd_merc_ask_up = {
	add_to_variable = { dd_merc_ask = 0.5 }
	clamp_variable = { var = dd_merc_ask min = 0.5 max = 50 }
}
dd_merc_ask_down = {
	subtract_from_variable = { dd_merc_ask = 0.5 }
	clamp_variable = { var = dd_merc_ask min = 0.5 max = 50 }
}

dd_merc_clear_buyer_pending = {
	clr_country_flag = dd_merc_pending
	clear_variable = dd_merc_pending_seller
}

dd_merc_reject_all_bids = {
	if = {
		limit = { has_variable = dd_merc_bid_0 }
		var:dd_merc_bid_0 = { dd_merc_clear_buyer_pending = yes }
		clear_variable = dd_merc_bid_0
	}
	if = {
		limit = { has_variable = dd_merc_bid_1 }
		var:dd_merc_bid_1 = { dd_merc_clear_buyer_pending = yes }
		clear_variable = dd_merc_bid_1
	}
	if = {
		limit = { has_variable = dd_merc_bid_2 }
		var:dd_merc_bid_2 = { dd_merc_clear_buyer_pending = yes }
		clear_variable = dd_merc_bid_2
	}
	if = {
		limit = { has_variable = dd_merc_bid_3 }
		var:dd_merc_bid_3 = { dd_merc_clear_buyer_pending = yes }
		clear_variable = dd_merc_bid_3
	}
}

dd_merc_compact_bids = {
	clear_array = dd_merc_bid_tmp
	if = {
		limit = { has_variable = dd_merc_bid_0 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_0 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_1 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_1 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_2 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_2 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_3 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_3 }
	}
	clear_variable = dd_merc_bid_0
	clear_variable = dd_merc_bid_1
	clear_variable = dd_merc_bid_2
	clear_variable = dd_merc_bid_3
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 0 } }
		set_variable = { dd_merc_bid_0 = dd_merc_bid_tmp^0 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 1 } }
		set_variable = { dd_merc_bid_1 = dd_merc_bid_tmp^1 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 2 } }
		set_variable = { dd_merc_bid_2 = dd_merc_bid_tmp^2 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 3 } }
		set_variable = { dd_merc_bid_3 = dd_merc_bid_tmp^3 }
	}
	clear_array = dd_merc_bid_tmp
}

dd_merc_request_clean = {
	# ROOT = buyer. dd_hire_tag = seller, dd_hire_price set by slot helper.
	if = {
		limit = {
			dd_merc_can_request = yes
			has_variable = dd_hire_tag
			NOT = { check_variable = { dd_hire_tag = THIS } }
			check_variable = { dd_hire_price > 0.49 }
			NOT = { check_variable = { treasury < dd_hire_price } }
		}
		var:dd_hire_tag = {
			if = {
				limit = { NOT = { has_variable = dd_merc_bid_0 } }
				set_variable = { dd_merc_bid_0 = ROOT }
				ROOT = {
					set_country_flag = dd_merc_pending
					set_variable = { dd_merc_pending_seller = PREV }
				}
			}
			else_if = {
				limit = { NOT = { has_variable = dd_merc_bid_1 } }
				set_variable = { dd_merc_bid_1 = ROOT }
				ROOT = {
					set_country_flag = dd_merc_pending
					set_variable = { dd_merc_pending_seller = PREV }
				}
			}
			else_if = {
				limit = { NOT = { has_variable = dd_merc_bid_2 } }
				set_variable = { dd_merc_bid_2 = ROOT }
				ROOT = {
					set_country_flag = dd_merc_pending

					set_country_flag = dd_merc_pending

dd_merc_deploy_to_buyer = {
	# ROOT = seller. dd_accept_buyer = buyer.
	if = {
		limit = {
			has_variable = dd_accept_buyer
			NOT = { has_country_flag = dd_merc_deployed }
			has_country_flag = dd_merc_offering
			check_variable = { dd_merc_ask > 0.49 }
			check_variable = { dd_merc_tpl > 0 }
			check_variable = { dd_merc_tpl < 8 }
		}
		var:dd_accept_buyer = {
			if = {
				limit = {
					has_country_flag = dd_merc_pending
					NOT = { has_country_flag = dd_merc_client }
					NOT = { check_variable = { treasury < ROOT.dd_merc_ask } }
				}
				subtract_from_variable = { treasury = ROOT.dd_merc_ask }
				ROOT = { add_to_variable = { treasury = ROOT.dd_merc_ask } }
				set_country_flag = dd_merc_client
				set_variable = { dd_merc_seller_tag = ROOT }
				dd_merc_clear_buyer_pending = yes
				give_military_access = ROOT
				ROOT = {
					give_military_access = PREV
					set_variable = { dd_merc_client_tag = PREV }
					set_variable = { dd_merc_hire_tpl = dd_merc_tpl }
					set_country_flag = dd_merc_deployed
					dd_merc_spawn_one = yes
				}
				every_enemy_country = {
					ROOT = {
						add_to_war = {
							targeted_alliance = PREV.PREV
							enemy = PREV
							hostility_reason = asked_to_join
						}
					}
				}
			}
		}
		dd_merc_reject_all_bids = yes
		dd_merc_clear_own_slot = yes
		clear_variable = dd_accept_buyer
	}
}
