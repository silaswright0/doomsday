# Mercenary market. Country-scoped. Deterministic. No random.
# Seller picks one division that is on the map.
# A hire request opens an accept/decline popup for the seller.
# Accept removes that division and raises the same template at the buyer's capital.
# Recall disbands that deployed division and raises it again at home.
# Buyer may rent only 1 merc division at a time.

# Template ids (shared catalog — matches locked 2026 OOBs):
# 1 Infantry Division | 2 Armored Division | 3 Mechanized Brigade
# 4 Marine Division | 5 Mountain Division | 6 Garrison Brigade | 7 Militia Brigade
# Deployed merc units use create_unit id = 77001 for recall.

# Stamp every on-map division with a stable id, then fill the offer list.
# Ids are never reused, so posting still points at the division you clicked.
dd_merc_refresh_templates = {
	if = {
		limit = { NOT = { has_variable = dd_merc_next_uid } }
		set_variable = { dd_merc_next_uid = 0 }
	}
	set_variable = { dd_c1 = 0 }
	set_variable = { dd_c2 = 0 }
	set_variable = { dd_c3 = 0 }
	set_variable = { dd_c4 = 0 }
	set_variable = { dd_c5 = 0 }
	set_variable = { dd_c6 = 0 }
	set_variable = { dd_c7 = 0 }
	set_variable = { dd_c8 = 0 }
	set_variable = { dd_merc_shown = 0 }
	clear_array = dd_merc_uids
	clear_array = dd_merc_kinds
	clear_array = dd_merc_ords
	clear_array = dd_merc_orgs
	clear_array = dd_merc_states
	every_country_division = {
		if = {
			limit = { NOT = { check_variable = { dd_merc_uid > 0 } } }
			ROOT = { add_to_variable = { dd_merc_next_uid = 1 } }
			set_variable = { dd_merc_uid = ROOT.dd_merc_next_uid }
		}
	}
	# List divisions from the state they are standing in, and copy that state
	# into the row at once. A variable saved as PREV keeps updating, so every
	# row was left pointing at the last state, Guantanamo Bay.
	every_state = {
		limit = {
			any_state_division = {
				OWNER = { tag = ROOT }
			}
		}
		every_state_division = {
			limit = {
				OWNER = { tag = ROOT }
				check_variable = { ROOT.dd_merc_shown < 80 }
			}
			if = {
				limit = {
					OR = {
						division_has_battalion_in_template = modern_armor
						division_has_battalion_in_template = light_armor
						division_has_battalion_in_template = medium_armor
						division_has_battalion_in_template = heavy_armor
						division_has_majority_template = light_armor
						division_has_majority_template = medium_armor
						division_has_majority_template = heavy_armor
					}
				}
				set_variable = { dd_merc_kind = 2 }
				ROOT = { add_to_variable = { dd_c2 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c2 }
			}
			else_if = {
				limit = {
					OR = {
						division_has_battalion_in_template = mechanized
						division_has_majority_template = mechanized
					}
				}
				set_variable = { dd_merc_kind = 3 }
				ROOT = { add_to_variable = { dd_c3 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c3 }
			}
			else_if = {
				limit = {
					OR = {
						division_has_battalion_in_template = marine
						division_has_majority_template = marine
					}
				}
				set_variable = { dd_merc_kind = 4 }
				ROOT = { add_to_variable = { dd_c4 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c4 }
			}
			else_if = {
				limit = {
					OR = {
						division_has_battalion_in_template = mountaineers
						division_has_majority_template = mountaineers
					}
				}
				set_variable = { dd_merc_kind = 5 }
				ROOT = { add_to_variable = { dd_c5 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c5 }
			}
			else_if = {
				limit = {
					division_has_majority_template = infantry
					division_has_battalion_in_template = engineer
					NOT = { division_has_battalion_in_template = artillery }
				}
				set_variable = { dd_merc_kind = 6 }
				ROOT = { add_to_variable = { dd_c6 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c6 }
			}
			else_if = {
				limit = {
					division_has_majority_template = infantry
					NOT = { division_has_battalion_in_template = engineer }
					NOT = { division_has_battalion_in_template = artillery }
				}
				set_variable = { dd_merc_kind = 7 }
				ROOT = { add_to_variable = { dd_c7 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c7 }
			}
			else_if = {
				limit = {
					OR = {
						division_has_battalion_in_template = infantry
						division_has_majority_template = infantry
					}
				}
				set_variable = { dd_merc_kind = 1 }
				ROOT = { add_to_variable = { dd_c1 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c1 }
			}
			else = {
				set_variable = { dd_merc_kind = 8 }
				ROOT = { add_to_variable = { dd_c8 = 1 } }
				set_variable = { dd_merc_ord = ROOT.dd_c8 }
			}
			if = {
				limit = { unit_organization > 0.66 }
				set_variable = { dd_merc_org_band = 2 }
			}
			else_if = {
				limit = { unit_organization > 0.33 }
				set_variable = { dd_merc_org_band = 1 }
			}
			else = {
				set_variable = { dd_merc_org_band = 0 }
			}
			set_temp_variable = { t_uid = dd_merc_uid }
			set_temp_variable = { t_kind = dd_merc_kind }
			set_temp_variable = { t_ord = dd_merc_ord }
			set_temp_variable = { t_org = dd_merc_org_band }
			ROOT = {
				add_to_array = { dd_merc_uids = t_uid }
				add_to_array = { dd_merc_kinds = t_kind }
				add_to_array = { dd_merc_ords = t_ord }
				add_to_array = { dd_merc_orgs = t_org }
				add_to_variable = { dd_merc_shown = 1 }
			}
			PREV = {
				ROOT = { add_to_array = { dd_merc_states = PREV } }
			}
		}
	}
	if = {
		limit = { check_variable = { dd_merc_pick_uid > 0 } }
		clr_country_flag = dd_merc_pick_ok
		every_country_division = {
			limit = { check_variable = { dd_merc_uid = PREV.dd_merc_pick_uid } }
			PREV = { set_country_flag = dd_merc_pick_ok }
		}
		if = {
			limit = { NOT = { has_country_flag = dd_merc_pick_ok } }
			set_variable = { dd_merc_pick_uid = 0 }
			clear_variable = dd_merc_tpl
			clear_variable = dd_merc_sel_kind
		}
		clr_country_flag = dd_merc_pick_ok
	}
}

dd_merc_ask_up = {
	add_to_variable = { dd_merc_ask = 0.5 }
	clamp_variable = { var = dd_merc_ask min = 0.5 max = 50 }
}
dd_merc_ask_down = {
	subtract_from_variable = { dd_merc_ask = 0.5 }
	clamp_variable = { var = dd_merc_ask min = 0.5 max = 50 }
}

	clr_country_flag = dd_merc_pending
	clear_variable = dd_merc_pending_seller
}

dd_merc_reject_all_bids = {
	if = {
		limit = { has_variable = dd_merc_bid_0 }
		var:dd_merc_bid_0 = {
			if = {
				limit = { has_country_flag = dd_merc_pending }
				country_event = { id = doomsday_merc.3 }
			}
			dd_merc_clear_buyer_pending = yes
		}
		clear_variable = dd_merc_bid_0
	}
	if = {
		limit = { has_variable = dd_merc_bid_1 }
		var:dd_merc_bid_1 = {
			if = {
				limit = { has_country_flag = dd_merc_pending }
				country_event = { id = doomsday_merc.3 }
			}
			dd_merc_clear_buyer_pending = yes
		}
		clear_variable = dd_merc_bid_1
	}
	if = {
		limit = { has_variable = dd_merc_bid_2 }
		var:dd_merc_bid_2 = {
			if = {
				limit = { has_country_flag = dd_merc_pending }
				country_event = { id = doomsday_merc.3 }
			}
			dd_merc_clear_buyer_pending = yes
		}
		clear_variable = dd_merc_bid_2
	}
	if = {
		limit = { has_variable = dd_merc_bid_3 }
		var:dd_merc_bid_3 = {
			if = {
				limit = { has_country_flag = dd_merc_pending }
				country_event = { id = doomsday_merc.3 }
			}
			dd_merc_clear_buyer_pending = yes
		}
		clear_variable = dd_merc_bid_3
	}
}

dd_merc_compact_bids = {
	clear_array = dd_merc_bid_tmp
	if = {
		limit = { has_variable = dd_merc_bid_0 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_0 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_1 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_1 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_2 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_2 }
	}
	if = {
		limit = { has_variable = dd_merc_bid_3 }
		add_to_array = { dd_merc_bid_tmp = dd_merc_bid_3 }
	}
	clear_variable = dd_merc_bid_0
	clear_variable = dd_merc_bid_1
	clear_variable = dd_merc_bid_2
	clear_variable = dd_merc_bid_3
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 0 } }
		set_variable = { dd_merc_bid_0 = dd_merc_bid_tmp^0 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 1 } }
		set_variable = { dd_merc_bid_1 = dd_merc_bid_tmp^1 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 2 } }
		set_variable = { dd_merc_bid_2 = dd_merc_bid_tmp^2 }
	}
	if = {
		limit = { check_variable = { dd_merc_bid_tmp^num > 3 } }
		set_variable = { dd_merc_bid_3 = dd_merc_bid_tmp^3 }
	}
	clear_array = dd_merc_bid_tmp
}

dd_merc_request_clean = {
	# ROOT = buyer. dd_hire_tag = seller, dd_hire_price set by slot helper.
	if = {
		limit = {
			dd_merc_can_request = yes
			has_variable = dd_hire_tag
			NOT = { check_variable = { dd_hire_tag = THIS } }
			check_variable = { dd_hire_price > 0.49 }
			NOT = { check_variable = { treasury < dd_hire_price } }
		}
		var:dd_hire_tag = {
			if = {
				limit = { NOT = { dd_merc_has_listed_division = yes } }
				ROOT = { country_event = { id = doomsday_merc.4 } }
			}
			else_if = {
				limit = { NOT = { has_variable = dd_merc_bid_0 } }
				set_variable = { dd_merc_bid_0 = ROOT }

# Deleted with disband after the convert, which returns that one division to the stockpile.

	if = {
		limit = {
			check_variable = { dd_merc_listed_tpl = 1 }
			NOT = { has_template = "DD Merc Hold 1" }
		}
		division_template = {
			name = "DD Merc Hold 1"

# Take the first on-map division of the listed template off the map and refund its gear.
# dd_merc_pulled stays set only if the division count actually fell.
dd_merc_pull_one = {
	clr_country_flag = dd_merc_pulled
	if = {
		limit = { dd_merc_has_listed_division = yes }
		set_variable = { dd_merc_divs_before = num_divisions }
		dd_merc_ensure_hold = yes
		every_country_division = {
		limit = {
			NOT = { ROOT = { has_country_flag = dd_merc_pulled } }
			check_variable = { dd_merc_uid = ROOT.dd_merc_listed_uid }
		}
		ROOT = { set_country_flag = dd_merc_pulled }
		if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 1 } }
			change_division_template = { division_template = "DD Merc Hold 1" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 2 } }
			change_division_template = { division_template = "DD Merc Hold 2" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 3 } }
			change_division_template = { division_template = "DD Merc Hold 3" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 4 } }
			change_division_template = { division_template = "DD Merc Hold 4" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 5 } }
			change_division_template = { division_template = "DD Merc Hold 5" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 6 } }
			change_division_template = { division_template = "DD Merc Hold 6" }
		}
		else_if = {
			limit = { check_variable = { ROOT.dd_merc_listed_tpl = 7 } }
			change_division_template = { division_template = "DD Merc Hold 7" }
		}
	}
	if = {
		limit = { has_country_flag = dd_merc_pulled }
		if = {
			limit = { check_variable = { dd_merc_listed_tpl = 1 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 1" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 2 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 2" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 3 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 3" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 4 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 4" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 5 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 5" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 6 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 6" disband = yes }
		}
		else_if = {
			limit = { check_variable = { dd_merc_listed_tpl = 7 } }
			delete_unit_template_and_units = { division_template = "DD Merc Hold 7" disband = yes }
		}
		if = {
			limit = { NOT = { check_variable = { num_divisions < dd_merc_divs_before } } }
			clr_country_flag = dd_merc_pulled
		}
		clear_variable = dd_merc_divs_before
	}

dd_merc_drop_buyer = {

	if = {
		limit = { has_variable = dd_drop_buyer }
		if = {
			limit = { check_variable = { dd_merc_bid_0 = dd_drop_buyer } }
			clear_variable = dd_merc_bid_0
		}
		if = {
			limit = { check_variable = { dd_merc_bid_1 = dd_drop_buyer } }
			clear_variable = dd_merc_bid_1
		}
		if = {
			limit = { check_variable = { dd_merc_bid_2 = dd_drop_buyer } }

			clear_variable = dd_merc_bid_3
		}

		clear_variable = dd_drop_buyer
	}
}

dd_merc_decline_from = {
	# Seller event option. FROM = buyer. No charge.
	# A second click after the request was already closed does nothing.
	if = {
		limit = { FROM = { has_country_flag = dd_merc_pending } }
		if = {
			limit = { NOT = { dd_merc_has_listed_division = yes } }
			FROM = { country_event = { id = doomsday_merc.4 } }
		}
		else = {
			FROM = { country_event = { id = doomsday_merc.3 } }
		}
		set_variable = { dd_drop_buyer = FROM }
		dd_merc_drop_buyer = yes
	}
}

dd_merc_deploy_to_buyer = {
	# ROOT = seller. dd_accept_buyer = buyer.
	if = {
		limit = {
			has_variable = dd_accept_buyer
			NOT = { has_country_flag = dd_merc_deployed }
			has_country_flag = dd_merc_offering
			check_variable = { dd_merc_listed_tpl > 0 }
			check_variable = { dd_merc_listed_tpl < 8 }
			check_variable = { dd_merc_ask > 0.49 }
		}
		clr_country_flag = dd_merc_deal_done
		var:dd_accept_buyer = {
			if = {
				limit = {
					has_country_flag = dd_merc_pending
					NOT = { has_country_flag = dd_merc_client }
					NOT = { check_variable = { treasury < ROOT.dd_merc_ask } }
				}
				ROOT = { dd_merc_pull_one = yes }
				if = {
					limit = { ROOT = { has_country_flag = dd_merc_pulled } }
					subtract_from_variable = { treasury = ROOT.dd_merc_ask }
					ROOT = { add_to_variable = { treasury = ROOT.dd_merc_ask } }
					set_country_flag = dd_merc_client
					set_variable = { dd_merc_seller_tag = ROOT }
					dd_merc_clear_buyer_pending = yes
					give_military_access = ROOT
					ROOT = {
						give_military_access = PREV
						set_variable = { dd_merc_client_tag = PREV }
						set_variable = { dd_merc_hire_tpl = dd_merc_listed_tpl }
						set_country_flag = dd_merc_deployed
						set_country_flag = dd_merc_deal_done
						dd_merc_spawn_one = yes
					}
					every_enemy_country = {
						ROOT = {
							add_to_war = {
								targeted_alliance = PREV.PREV
								enemy = PREV
								hostility_reason = asked_to_join
							}
						}
					}
					country_event = { id = doomsday_merc.2 }
				}
				else = {
					country_event = { id = doomsday_merc.4 }
					ROOT = {
						set_variable = { dd_drop_buyer = PREV }
						dd_merc_drop_buyer = yes
					}
				}
			}
		}
		if = {
			limit = { has_country_flag = dd_merc_deal_done }
			dd_merc_reject_all_bids = yes
			dd_merc_clear_own_slot = yes
			clr_country_flag = dd_merc_deal_done
			clr_country_flag = dd_merc_pulled
		}
		clear_variable = dd_accept_buyer
	}
}
