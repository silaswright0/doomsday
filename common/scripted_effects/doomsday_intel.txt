# Intelligence agencies. Country-scoped. Deterministic (no random_list).
#
# Insurgent tag policy:
# - Arm Insurgents → reserved off-map pool IG1–IG4 (cosmetic DD_INSURGENT).
#   Slots recycle when the IG has no states and is not at war.
# - Coup civil war → vanilla start_civil_war (dynamic sibling of the target,
#   ~25% army/states). Cosmetic DD_COUP_REB. Not taken from the IG pool.

dd_intel_init = {
	if = {
		limit = { NOT = { has_country_flag = dd_intel_init } }
		set_country_flag = dd_intel_init
		set_variable = { dd_intel_mission = 0 }
		if = {
			limit = { tag = USA }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_cia
		}
		else_if = {
			limit = { tag = SOV }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_svr
		}
		else_if = {
			limit = { tag = ISR }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_mossad
		}
		else_if = {
			limit = { tag = CHI }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_mss
		}
		else_if = {
			limit = { tag = DPK }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_rgb
		}
		else_if = {
			limit = { tag = PER }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_vaja
		}
		else_if = {
			limit = { tag = FRA }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_dgse
		}
		else_if = {
			limit = { tag = ENG }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_mi6
		}
		else = {
			set_variable = { dd_intel_tier = 0 }
		}
	}
}

dd_intel_yearly_cost = {
	if = {
		limit = { check_variable = { dd_intel_tier = 1 } }
		if = {
			limit = { NOT = { check_variable = { treasury < 1.5 } } }
			subtract_from_variable = { treasury = 1.5 }
		}
	}
	else_if = {
		limit = { check_variable = { dd_intel_tier = 2 } }
		if = {
			limit = { NOT = { check_variable = { treasury < 0.5 } } }
			subtract_from_variable = { treasury = 0.5 }
		}
	}
}

dd_intel_start_cooldown = {
	set_variable = { dd_intel_cd = 12 }
	set_country_flag = dd_intel_mission_cd
}

dd_intel_cd_tick = {
	if = {
		limit = { check_variable = { dd_intel_cd > 0 } }
		subtract_from_variable = { dd_intel_cd = 1 }
		if = {
			limit = { NOT = { check_variable = { dd_intel_cd > 0 } } }
			set_variable = { dd_intel_cd = 0 }
			clr_country_flag = dd_intel_mission_cd
		}
	}
	else = {
		# Drop leftover valued flags from the old days=365 value=1 cooldown.
		clr_country_flag = dd_intel_mission_cd
		set_variable = { dd_intel_cd = 0 }
	}
}

dd_intel_cancel_select_effect = {
	set_variable = { dd_intel_mission = 0 }
	clear_variable = dd_intel_sabo_country
	clear_variable = dd_intel_mission_target
	clr_country_flag = dd_intel_locked
	clear_array = dd_intel_cmd_pool
}

dd_intel_finish_select = {
	dd_intel_start_cooldown = yes
	dd_intel_cancel_select_effect = yes
}

# After country lock: cooldown already running; clear mission + pick state.
dd_intel_finish_locked = {
	clr_country_flag = dd_intel_locked
	clear_variable = dd_intel_mission_target
	clear_array = dd_intel_cmd_pool
	set_variable = { dd_intel_mission = 0 }
	clear_variable = dd_intel_sabo_country
}

dd_intel_lock_assassinate_country = {
	# Commit: charge, cooldown, no cancel. Then pick a commander.
	subtract_from_variable = { treasury = 3.0 }
	dd_intel_start_cooldown = yes
	set_variable = { dd_intel_mission_target = FROM }
	set_country_flag = dd_intel_locked
	clear_array = dd_intel_cmd_pool
	FROM = {
		every_army_leader = {
			add_to_array = { ROOT.dd_intel_cmd_pool = THIS }
		}
	}
}

dd_intel_lock_steal_country = {
	subtract_from_variable = { treasury = 2.0 }
	dd_intel_start_cooldown = yes
	set_variable = { dd_intel_mission_target = FROM }
	set_country_flag = dd_intel_locked
}

dd_intel_assassinate_leader = {
	# FROM = unit leader from target_array. 80% success (d10 0-7).
	# Roll from ROOT + locked country (FROM is a leader, not a country).
	set_temp_variable = { dd_intel_roll = num_owned_states }
	add_to_temp_variable = { dd_intel_roll = num_divisions }
	add_to_temp_variable = { dd_intel_roll = political_power }
	add_to_temp_variable = { dd_intel_roll = treasury }
	var:dd_intel_mission_target = {
		add_to_temp_variable = { PREV.dd_intel_roll = num_owned_states }
		add_to_temp_variable = { PREV.dd_intel_roll = num_divisions }
	}
	modulo_temp_variable = { dd_intel_roll = 10 }
	if = {
		limit = { check_variable = { dd_intel_roll < 8 } }
		FROM = { remove_unit_leader = yes }
	}
	dd_intel_finish_locked = yes
}

	if = {
		limit = { NOT = { has_country_flag = dd_intel_militia_ready } }
		set_country_flag = dd_intel_militia_ready
		division_template = {
			name = "Insurgent Militia"
			is_locked = yes
			regiments = {
				militia = { x = 0 y = 0 }
				militia = { x = 0 y = 1 }
				militia = { x = 1 y = 0 }
				militia = { x = 1 y = 1 }
				militia = { x = 2 y = 0 }
				militia = { x = 2 y = 1 }
			}
		}
	}
}

dd_intel_pick_ig = {
	# Claim first free reserved insurgent slot into dd_intel_ig_picked.
	clear_variable = dd_intel_ig_picked
	if = {
		limit = { IG1 = { dd_ig_is_free = yes } }
		set_variable = { dd_intel_ig_picked = IG1 }
	}
	else_if = {
		limit = { IG2 = { dd_ig_is_free = yes } }
		set_variable = { dd_intel_ig_picked = IG2 }
	}
	else_if = {
		limit = { IG3 = { dd_ig_is_free = yes } }
		set_variable = { dd_intel_ig_picked = IG3 }
	}
	else_if = {
		limit = { IG4 = { dd_ig_is_free = yes } }
		set_variable = { dd_intel_ig_picked = IG4 }
	}
}

dd_intel_spawn_insurgents = {
	# Activate a reserved IG tag with 2×6w militia in the target capital.
	dd_intel_pick_ig = yes
	if = {
		limit = { has_variable = dd_intel_ig_picked }
		var:dd_intel_ig_picked = {
			clr_country_flag = dd_ig_dormant
			set_country_flag = dd_ig_active
			set_cosmetic_tag = DD_INSURGENT
			set_politics = {
				ruling_party = left_wing_nationalism
				elections_allowed = no
			}
			set_popularities = {
				left_wing_nationalism = 100
			}
			dd_intel_ensure_militia = yes
			declare_war_on = {
				target = var:ROOT.dd_intel_insurgent_target
				type = annex_everything
			}
			var:ROOT.dd_intel_insurgent_target = {
				add_timed_idea = { idea = dd_intel_insurgency days = 180 }
				capital_scope = {
					create_unit = {
						division = "name = \"1st Insurgent Militia\" division_template = \"Insurgent Militia\" start_experience_factor = 0.15 start_equipment_factor = 0.8"
						owner = PREV.PREV
						count = 1
						allow_spawning_on_enemy_provs = yes
					}
					create_unit = {
						division = "name = \"2nd Insurgent Militia\" division_template = \"Insurgent Militia\" start_experience_factor = 0.15 start_equipment_factor = 0.8"
						owner = PREV.PREV
						count = 1
						allow_spawning_on_enemy_provs = yes
					}
				}
			}
		}
	}
	clear_variable = dd_intel_insurgent_target
	clear_variable = dd_intel_ig_picked
}

dd_ig_monthly_cleanup = {
	# Recycle reserved IG slots after the uprising is over.
	if = {
		limit = {
			OR = { tag = IG1 tag = IG2 tag = IG3 tag = IG4 }
			has_country_flag = dd_ig_active
			num_owned_states < 1
			NOT = { has_war = yes }
		}
		delete_unit = {
			disband = yes
		}
		drop_cosmetic_tag = yes
		clr_country_flag = dd_ig_active
		set_country_flag = dd_ig_dormant
	}
}

dd_intel_assassinate_one = {
	# Deterministic: remove the first army leader (general or field marshal).
	# remove_unit_leader is the reliable unit-leader kill; retire alone can no-op.
	clr_country_flag = dd_intel_kill_done
	FROM = {
		every_army_leader = {
			limit = { NOT = { ROOT = { has_country_flag = dd_intel_kill_done } } }
			remove_unit_leader = yes
			ROOT = { set_country_flag = dd_intel_kill_done }
		}
	}
	clr_country_flag = dd_intel_kill_done
}
