# Intelligence agencies. Country-scoped. Deterministic (no random_list).
#
# Insurgent tag policy:
# - Arm Insurgents → reserved off-map pool IG1–IG4 (cosmetic DD_INSURGENT).
#   Slots recycle when the IG has no states and is not at war.
# - Coup civil war → vanilla start_civil_war (dynamic sibling of the target,
#   ~25% army/states). Cosmetic DD_COUP_REB. Not taken from the IG pool.

dd_intel_init = {
	if = {
		limit = { NOT = { has_country_flag = dd_intel_init } }
		set_country_flag = dd_intel_init
		set_variable = { dd_intel_mission = 0 }
		if = {
			limit = { tag = USA }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_cia
		}
		else_if = {
			limit = { tag = SOV }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_svr
		}
		else_if = {
			limit = { tag = ISR }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_mossad
		}
		else_if = {
			limit = { tag = CHI }
			set_variable = { dd_intel_tier = 1 }
			add_ideas = dd_intel_mss
		}
		else_if = {
			limit = { tag = DPK }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_rgb
		}
		else_if = {
			limit = { tag = PER }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_vaja
		}
		else_if = {
			limit = { tag = FRA }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_dgse
		}
		else_if = {
			limit = { tag = ENG }
			set_variable = { dd_intel_tier = 2 }
			add_ideas = dd_intel_mi6
		}
		else = {
			set_variable = { dd_intel_tier = 0 }
		}
	}
}

dd_intel_yearly_cost = {
	if = {
		limit = { check_variable = { dd_intel_tier = 1 } }
		if = {
			limit = { NOT = { check_variable = { treasury < 1.5 } } }
			subtract_from_variable = { treasury = 1.5 }
		}
	}
	else_if = {
		limit = { check_variable = { dd_intel_tier = 2 } }
		if = {
			limit = { NOT = { check_variable = { treasury < 0.5 } } }
			subtract_from_variable = { treasury = 0.5 }
		}
	}
}

dd_intel_start_cooldown = {
	set_variable = { dd_intel_cd = 12 }
	set_country_flag = dd_intel_mission_cd
}

dd_intel_cd_tick = {
	if = {
		limit = { check_variable = { dd_intel_cd > 0 } }
		subtract_from_variable = { dd_intel_cd = 1 }
		if = {
			limit = { NOT = { check_variable = { dd_intel_cd > 0 } } }
			set_variable = { dd_intel_cd = 0 }
			clr_country_flag = dd_intel_mission_cd
		}
	}
	else = {
		# Drop leftover valued flags from the old days=365 value=1 cooldown.
		clr_country_flag = dd_intel_mission_cd
		set_variable = { dd_intel_cd = 0 }
	}
}

dd_intel_cancel_select_effect = {
	# Release IG reservation if aborting arm before timer (mission cleared without arm_running).
	if = {
		limit = {
			has_variable = dd_intel_ig_picked
			NOT = { has_country_flag = dd_intel_arm_running }
		}
		var:dd_intel_ig_picked = { clr_country_flag = dd_ig_reserved }
		clear_variable = dd_intel_ig_picked
	}
	set_variable = { dd_intel_mission = 0 }
	clear_variable = dd_intel_sabo_country
	clear_variable = dd_intel_mission_target
	clr_country_flag = dd_intel_locked
	clr_country_flag = dd_intel_kill_success
	clr_country_flag = dd_show_cmd_picker
	clear_array = dd_intel_cmd_pool
	set_variable = { dd_intel_cmd_idx = 0 }
}

dd_intel_finish_select = {
	dd_intel_start_cooldown = yes
	dd_intel_cancel_select_effect = yes
}

# After country lock: cooldown already running; clear mission + pick state.
dd_intel_finish_locked = {
	clr_country_flag = dd_intel_locked
	clr_country_flag = dd_show_cmd_picker
	clear_variable = dd_intel_mission_target
	clear_array = dd_intel_cmd_pool
	set_variable = { dd_intel_cmd_idx = 0 }
	set_variable = { dd_intel_mission = 0 }
	clear_variable = dd_intel_sabo_country
	clr_country_flag = dd_intel_kill_success
}

# If target was self, open blame map-click phase (mission 8).
# FROM is the clicked state.
dd_intel_maybe_start_blame = {
	FROM = {
		owner = {
			if = {
				limit = { tag = ROOT }
				ROOT = {
					set_variable = { dd_intel_mission = 8 }
					country_event = { id = doomsday_intel.20 hours = 1 }
				}
			}
		}
	}
}

dd_intel_lock_assassinate_country = {
	subtract_from_variable = { treasury = 3.0 }
	dd_intel_start_cooldown = yes
	set_country_flag = dd_intel_locked
	set_country_flag = dd_show_cmd_picker
	set_variable = { dd_intel_cmd_idx = 0 }
	clear_array = dd_intel_cmd_pool
	FROM = {
		owner = {
			ROOT = { set_variable = { dd_intel_mission_target = PREV } }
			every_army_leader = {
				add_to_array = { ROOT.dd_intel_cmd_pool = THIS }
			}
		}
	}
	dd_intel_cmd_sync_stats = yes
}

dd_intel_lock_steal_country = {
	subtract_from_variable = { treasury = 2.0 }
	dd_intel_start_cooldown = yes
	FROM = {
		owner = {
			ROOT = { set_variable = { dd_intel_mission_target = PREV } }
		}
	}
	set_country_flag = dd_intel_locked
}

# Export selected commander skill lines for the picker GUI (deterministic).
dd_intel_cmd_sync_stats = {
	set_variable = { dd_intel_cmd_skill = 0 }
	set_variable = { dd_intel_cmd_atk = 0 }
	set_variable = { dd_intel_cmd_def = 0 }
	set_variable = { dd_intel_cmd_pln = 0 }
	set_variable = { dd_intel_cmd_log = 0 }
	if = {
		limit = { check_variable = { dd_intel_cmd_pool^num > 0 } }
		var:dd_intel_cmd_pool^dd_intel_cmd_idx = {
			set_variable = { ROOT.dd_intel_cmd_skill = skill }
			set_variable = { ROOT.dd_intel_cmd_atk = attack_skill_level }
			set_variable = { ROOT.dd_intel_cmd_def = defense_skill_level }
			set_variable = { ROOT.dd_intel_cmd_pln = planning_skill_level }
			set_variable = { ROOT.dd_intel_cmd_log = logistics_skill_level }
		}
	}
}

dd_intel_cmd_prev = {
	if = {
		limit = { check_variable = { dd_intel_cmd_pool^num > 0 } }
		if = {
			limit = { check_variable = { dd_intel_cmd_idx > 0 } }
			subtract_from_variable = { dd_intel_cmd_idx = 1 }
		}
		else = {
			set_variable = { dd_intel_cmd_idx = dd_intel_cmd_pool^num }
			subtract_from_variable = { dd_intel_cmd_idx = 1 }
		}
		dd_intel_cmd_sync_stats = yes
	}
}

dd_intel_cmd_next = {
	if = {
		limit = { check_variable = { dd_intel_cmd_pool^num > 0 } }
		add_to_variable = { dd_intel_cmd_idx = 1 }
		if = {
			limit = { NOT = { check_variable = { dd_intel_cmd_idx < dd_intel_cmd_pool^num } } }
			set_variable = { dd_intel_cmd_idx = 0 }
		}
		dd_intel_cmd_sync_stats = yes
	}
}

# Abort after country lock (cash already spent / CD running).
dd_intel_cmd_cancel = {
	dd_intel_finish_locked = yes
}

dd_intel_assassinate_confirm = {
	# 80% success (d10 0-7). Kill selected commander from pool.
	clr_country_flag = dd_intel_kill_success
	set_temp_variable = { dd_intel_roll = num_owned_states }
	add_to_temp_variable = { dd_intel_roll = num_divisions }
	add_to_temp_variable = { dd_intel_roll = political_power }
	add_to_temp_variable = { dd_intel_roll = treasury }
	var:dd_intel_mission_target = {
		add_to_temp_variable = { PREV.dd_intel_roll = num_owned_states }
		add_to_temp_variable = { PREV.dd_intel_roll = num_divisions }
	}
	modulo_temp_variable = { dd_intel_roll = 10 }
	if = {
		limit = {
			check_variable = { dd_intel_roll < 8 }
			check_variable = { dd_intel_cmd_pool^num > 0 }
		}
		set_temp_variable = { dd_kill_leader = dd_intel_cmd_pool^dd_intel_cmd_idx }
		var:dd_kill_leader = {
			remove_unit_leader = yes
		}
		set_country_flag = dd_intel_kill_success
	}
	# News + optional blame use mission target as FROM.
	var:dd_intel_mission_target = {
		if = {
			limit = { ROOT = { has_country_flag = dd_intel_kill_success } }
			news_event = { id = doomsday_intel.3 hours = 2 }
		}
		else = {
			news_event = { id = doomsday_intel.4 hours = 2 }
		}
	}
	if = {
		limit = { var:dd_intel_mission_target = { tag = ROOT } }
		set_variable = { dd_intel_mission = 8 }
		clr_country_flag = dd_intel_locked
		clr_country_flag = dd_show_cmd_picker
		clear_array = dd_intel_cmd_pool
		country_event = { id = doomsday_intel.20 hours = 1 }
	}
	else = {
		dd_intel_finish_locked = yes
	}
	clr_country_flag = dd_intel_kill_success
}

dd_intel_apply_blame = {
	# FROM = clicked capital. ROOT = blamer.
	FROM = {
		owner = {
			ROOT = {
				add_opinion_modifier = {
					target = PREV
					modifier = dd_intel_blame_hatred
				}
			}
			add_opinion_modifier = {
				target = ROOT
				modifier = dd_intel_blame_hatred
			}
		}
	}
	add_timed_idea = { idea = dd_intel_blame_ws days = 730 }
	set_variable = { dd_intel_mission = 0 }
}

dd_intel_arm_confirm = {
	# FROM = clicked capital. Reserve IG now so spawn cannot silently no-op at day 180.
	dd_intel_pick_ig = yes
	if = {
		limit = { has_variable = dd_intel_ig_picked }
		subtract_from_variable = { treasury = 2.5 }
		dd_intel_start_cooldown = yes
		FROM = {
			owner = {
				ROOT = { set_variable = { dd_intel_insurgent_target = PREV } }
			}
		}
		set_country_flag = dd_intel_arm_running
		var:dd_intel_ig_picked = {
			set_country_flag = dd_ig_reserved
		}
		set_variable = { dd_intel_mission = 0 }
		activate_mission = dd_intel_arm_countdown
	}
	else = {
		# Pool emptied between click and confirm — abort select, no charge.
		set_variable = { dd_intel_mission = 0 }
	}
}

dd_intel_spawn_insurgents = {
	# Use IG reserved at confirm. If target/IG gone, collapse + release.
	clr_country_flag = dd_intel_arm_ok
	if = {
		limit = {
			has_variable = dd_intel_ig_picked
			has_variable = dd_intel_insurgent_target
			var:dd_intel_insurgent_target = { exists = yes }
		}
		var:dd_intel_ig_picked = {
			clr_country_flag = dd_ig_dormant
			clr_country_flag = dd_ig_reserved
			set_country_flag = dd_ig_active
			set_cosmetic_tag = DD_INSURGENT
			set_politics = {
				ruling_party = left_wing_nationalism
				elections_allowed = no
			}
			set_popularities = {
				left_wing_nationalism = 100
			}
			dd_intel_ensure_militia = yes
			declare_war_on = {
				target = var:ROOT.dd_intel_insurgent_target
				type = annex_everything
			}
			var:ROOT.dd_intel_insurgent_target = {
				add_timed_idea = { idea = dd_intel_insurgency days = 180 }
				capital_scope = {
					create_unit = {
						division = "name = \"1st Insurgent Militia\" division_template = \"Insurgent Militia\" start_experience_factor = 0.15 start_equipment_factor = 0.8"
						owner = PREV.PREV
						count = 1
						allow_spawning_on_enemy_provs = yes
					}
					create_unit = {
						division = "name = \"2nd Insurgent Militia\" division_template = \"Insurgent Militia\" start_experience_factor = 0.15 start_equipment_factor = 0.8"
						owner = PREV.PREV
						count = 1
						allow_spawning_on_enemy_provs = yes
					}
				}
			}
		}
		ROOT = { set_country_flag = dd_intel_arm_ok }
		var:dd_intel_insurgent_target = {
			news_event = { id = doomsday_intel.5 hours = 2 }
		}
		if = {
			limit = { var:dd_intel_insurgent_target = { tag = ROOT } }
			set_variable = { dd_intel_mission = 8 }
			country_event = { id = doomsday_intel.20 hours = 1 }
		}
	}
	else = {
		if = {
			limit = { has_variable = dd_intel_ig_picked }
			var:dd_intel_ig_picked = {
				clr_country_flag = dd_ig_reserved
			}
		}
		if = {
			limit = { has_variable = dd_intel_insurgent_target }
			var:dd_intel_insurgent_target = {
				news_event = { id = doomsday_intel.6 hours = 2 }
			}
		}
		else = {
			news_event = { id = doomsday_intel.6 hours = 2 }
		}
	}
	clr_country_flag = dd_intel_arm_ok
	clear_variable = dd_intel_insurgent_target
	clear_variable = dd_intel_ig_picked
}

dd_ig_monthly_cleanup = {
	# Recycle reserved / active IG slots after the uprising is over.
	if = {
		limit = {
			OR = { tag = IG1 tag = IG2 tag = IG3 tag = IG4 }
			OR = {
				has_country_flag = dd_ig_active
				has_country_flag = dd_ig_reserved
			}
			check_variable = { num_owned_states < 1 }
			NOT = { has_war = yes }
			# Reserved-but-waiting slots stay reserved until spawn/collapse.
			NOT = { has_country_flag = dd_ig_reserved }
		}
		delete_unit = {
			disband = yes
		}
		drop_cosmetic_tag = yes
		clr_country_flag = dd_ig_active
		clr_country_flag = dd_ig_reserved
		set_country_flag = dd_ig_dormant
	}
	# Stuck active+war with no units: white peace if target gone, else leave alone.
	if = {
		limit = {
			OR = { tag = IG1 tag = IG2 tag = IG3 tag = IG4 }
			has_country_flag = dd_ig_active
			check_variable = { num_owned_states < 1 }
			has_war = yes
			NOT = { any_enemy_country = { exists = yes } }
		}
		every_country = {
			limit = { has_war_with = PREV }
			white_peace = PREV
		}
	}
}

dd_intel_assassinate_one = {
	# Deterministic: remove the first army leader (general or field marshal).
	# remove_unit_leader is the reliable unit-leader kill; retire alone can no-op.
	clr_country_flag = dd_intel_kill_done
	FROM = {
		every_army_leader = {
			limit = { NOT = { ROOT = { has_country_flag = dd_intel_kill_done } } }
			remove_unit_leader = yes
			ROOT = { set_country_flag = dd_intel_kill_done }
		}
	}
	clr_country_flag = dd_intel_kill_done
}

dd_intel_coup_civil_war = {
	# 25% of the country (army + states) rises as a junta civil-war sibling.
	FROM = {
		owner = {
		add_popularity = {
			ideology = military_junta
			popularity = 0.25
		}
		start_civil_war = {
			ideology = military_junta
			size = 0.25
		}
		every_enemy_country = {
			limit = {
				original_tag = PREV
				has_government = military_junta
			}
			set_cosmetic_tag = DD_COUP_REB
			set_country_flag = dd_intel_coup_rebels
			set_variable = { dd_intel_tier = 0 }
		}
		news_event = { id = doomsday_intel.9 hours = 2 }
		}
	}
}

dd_intel_coup_resolve = {
	# d10: 0-2 success (30%), 3-7 fail (50%), 8-9 civil war (20%).
	dd_intel_det_d10 = yes
	if = {
		limit = { check_variable = { dd_intel_roll < 3 } }
		dd_intel_coup_success = yes
	}
	else_if = {
		limit = { check_variable = { dd_intel_roll < 8 } }
		dd_intel_coup_fail = yes
	}
	else = {
		dd_intel_coup_civil_war = yes
	}
	dd_intel_maybe_start_blame = yes
}
