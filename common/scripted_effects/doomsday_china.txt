dd_chi_paranoia_sync = {
	if = {
		limit = { check_variable = { dd_chi_paranoia < 0 } }
		set_variable = { dd_chi_paranoia = 0 }
	}
	if = {
		limit = { check_variable = { dd_chi_paranoia > 100 } }
		set_variable = { dd_chi_paranoia = 100 }
	}
	set_variable = { dd_chi_para_pp = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_pp = -0.001 }
	set_variable = { dd_chi_para_stab = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_stab = -0.0015 }
	set_variable = { dd_chi_para_fac = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_fac = -0.0008 }
	set_variable = { dd_chi_para_org = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_org = -0.001 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = dd_chi_paranoia } } }
		add_dynamic_modifier = { modifier = dd_chi_paranoia }
	}
	force_update_dynamic_modifier = yes
	set_variable = { dd_chi_para_meter = dd_chi_paranoia }
	add_to_variable = { dd_chi_para_meter = 2.499 }
	divide_variable = { dd_chi_para_meter = 5 }
	round_variable = dd_chi_para_meter
	add_to_variable = { dd_chi_para_meter = 1 }
	if = {
		limit = { check_variable = { dd_chi_para_meter < 1 } }
		set_variable = { dd_chi_para_meter = 1 }
	}
	if = {
		limit = { check_variable = { dd_chi_para_meter > 21 } }
		set_variable = { dd_chi_para_meter = 21 }
	}
	if = {
		limit = { NOT = { has_variable = dd_chi_confidence } }
		set_variable = { dd_chi_confidence = 100 }
	}
}

dd_chi_paranoia_month = {
	if = {
		limit = { NOT = { has_variable = dd_chi_paranoia } }
		set_variable = { dd_chi_paranoia = 25 }
	}
	if = {
		limit = { NOT = { has_country_flag = dd_chi_paranoia_halted } }
		if = {
			limit = { has_country_flag = dd_chi_paranoia_contained }
			add_to_variable = { dd_chi_paranoia = 1 }
		}
		else = {
			add_to_variable = { dd_chi_paranoia = 4 }
		}
	}
	if = {
		limit = {
			check_variable = { dd_chi_paranoia > 99.9 }
			NOT = { has_country_flag = dd_chi_security_seized }
		}
		set_country_flag = dd_chi_security_seized
		set_variable = { dd_chi_paranoia = 50 }
		add_stability = -0.10
		add_war_support = -0.05
		add_political_power = -75
		country_event = { id = dd_chi_para.1 }
	}
	dd_chi_paranoia_sync = yes
}

# Two passes, state id order. A state with two civilian factories can give both.
dd_chi_convert_civ_to_mil = {
	set_variable = { dd_chi_mills_left = 2 }
	every_owned_state = {
		limit = {
			is_controlled_by = ROOT
			industrial_complex > 0
			check_variable = { ROOT.dd_chi_mills_left > 0 }
		}
		remove_building = { type = industrial_complex level = 1 }
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
		ROOT = { add_to_variable = { dd_chi_mills_left = -1 } }
	}
	every_owned_state = {
		limit = {
			is_controlled_by = ROOT
			industrial_complex > 0
			check_variable = { ROOT.dd_chi_mills_left > 0 }
		}
		remove_building = { type = industrial_complex level = 1 }
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
		ROOT = { add_to_variable = { dd_chi_mills_left = -1 } }
	}
	set_variable = { dd_chi_mills_left = 0 }
}

# Cores on every state China starts with as a core. Claims on the states China only claims.
# Called in the splinter's scope.
dd_chi_grant_china_map = {
	save_event_target_as = dd_chi_splinter
	every_state = {
		limit = { is_core_of = CHI }
		add_core_of = event_target:dd_chi_splinter
	}
	every_state = {
		limit = {
			is_claimed_by = CHI
			NOT = { is_core_of = CHI }
		}
		add_claim_by = event_target:dd_chi_splinter
	}
}

# Low confidence, the three standing strains still open, and Taiwan lost.
dd_chi_regime_watch = {
	if = {
		limit = { has_war_with = FOR }
		set_country_flag = dd_chi_fought_taiwan
	}
	if = {
		limit = {
			has_country_flag = dd_chi_fought_taiwan
			NOT = { has_war_with = FOR }
			NOT = { owns_state = 524 }
		}
		set_country_flag = dd_chi_taiwan_failed
	}
	if = {
		limit = {
			NOT = { has_global_flag = dd_chi_civil_war_started }
			NOT = { has_country_flag = dd_chi_civil_war_pending }
			has_country_flag = dd_chi_taiwan_failed
			check_variable = { dd_chi_confidence < 25 }
			NOT = { has_completed_focus = CHI_real_estate_3 }
			NOT = { has_completed_focus = CHI_local_debt_3 }
			NOT = { has_completed_focus = CHI_youth_unemployment_3 }
		}
		set_country_flag = dd_chi_civil_war_pending
		country_event = { id = dd_chi_cw.1 }
	}
}

# Elected governments cannot keep an embargo on China while at peace.
# Checked once a month. Deterministic country order. No random.
dd_chi_lift_elected_embargoes = {
	if = {
		limit = { has_idea = dd_chi_alternative_hegemony }
		every_other_country = {
			limit = {
				dd_is_elected_gov = yes
				NOT = { has_war_with = ROOT }
				ROOT = { is_embargoed_by = PREV }
			}
			ROOT = { break_embargo = PREV }
		}
	}
}
