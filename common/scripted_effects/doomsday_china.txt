# Breakaways do not keep Beijing's paranoia or the officer penalty.
dd_chi_drop_paranoia = {
	if = {
		limit = { has_dynamic_modifier = { modifier = dd_chi_paranoia } }
		remove_dynamic_modifier = { modifier = dd_chi_paranoia }
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = dd_chi_confidence_fear } }
		remove_dynamic_modifier = { modifier = dd_chi_confidence_fear }
	}
}

dd_chi_paranoia_sync = {
	if = {
		limit = { NOT = { original_tag = CHI } }
		dd_chi_drop_paranoia = yes
	}
	else = {
	if = {
		limit = { check_variable = { dd_chi_paranoia < 0 } }
		set_variable = { dd_chi_paranoia = 0 }
	}
	if = {
		limit = { check_variable = { dd_chi_paranoia > 100 } }
		set_variable = { dd_chi_paranoia = 100 }
	}
	set_variable = { dd_chi_para_pp = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_pp = -0.001 }
	set_variable = { dd_chi_para_stab = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_stab = -0.0015 }
	set_variable = { dd_chi_para_fac = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_fac = -0.0008 }
	set_variable = { dd_chi_para_org = dd_chi_paranoia }
	multiply_variable = { dd_chi_para_org = -0.001 }
	if = {
		limit = {
			original_tag = CHI
			NOT = { has_country_flag = dd_chi_organs_sidelined }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = dd_chi_paranoia } } }
			add_dynamic_modifier = { modifier = dd_chi_paranoia }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_chi_paranoia } }
			remove_dynamic_modifier = { modifier = dd_chi_paranoia }
		}
	}
	set_variable = { dd_chi_para_meter = dd_chi_paranoia }
	add_to_variable = { dd_chi_para_meter = 2.499 }
	divide_variable = { dd_chi_para_meter = 5 }
	round_variable = dd_chi_para_meter
	add_to_variable = { dd_chi_para_meter = 1 }
	if = {
		limit = { check_variable = { dd_chi_para_meter < 1 } }
		set_variable = { dd_chi_para_meter = 1 }
	}
	if = {
		limit = { check_variable = { dd_chi_para_meter > 21 } }
		set_variable = { dd_chi_para_meter = 21 }
	}
	if = {
		limit = { NOT = { has_variable = dd_chi_confidence } }
		set_variable = { dd_chi_confidence = 100 }
	}
	dd_chi_sync_confidence = yes
	}
}

dd_chi_paranoia_month = {
	if = {
		limit = { has_country_flag = dd_chi_organs_sidelined }
		dd_chi_paranoia_sync = yes
	}
	else = {
		if = {
			limit = { NOT = { has_variable = dd_chi_paranoia } }
			set_variable = { dd_chi_paranoia = 50 }
		}
		if = {
			limit = { NOT = { has_country_flag = dd_chi_paranoia_halted } }
			if = {
				limit = { has_country_flag = dd_chi_paranoia_contained }
				add_to_variable = { dd_chi_paranoia = 1 }
			}
			else = {
				add_to_variable = { dd_chi_paranoia = 4 }
			}
			if = {
				limit = { has_idea = CHI_cai_qi }
				add_to_variable = { dd_chi_paranoia = -1 }
			}
			if = {
				limit = { has_idea = CHI_chen_yixin }
				add_to_variable = { dd_chi_paranoia = 2 }
			}
		}
		if = {
			limit = {
				check_variable = { dd_chi_paranoia > 99.9 }
				NOT = { has_country_flag = dd_chi_security_seized }
			}
			set_country_flag = dd_chi_security_seized
			set_variable = { dd_chi_paranoia = 50 }
			add_stability = -0.10
			add_war_support = -0.05
			add_political_power = -75
			country_event = { id = dd_chi_para.1 }
		}
		dd_chi_paranoia_sync = yes
	}
}

# New Year. Safe if China is gone, the meter is gone, or confidence is already full.

		dd_chi_sync_confidence = yes
	}

# Full confidence has no officer penalty. Each purge cuts 25, and the penalty scales with what is missing.
dd_chi_sync_confidence = {
	if = {
		limit = { NOT = { has_variable = dd_chi_confidence } }
		set_variable = { dd_chi_confidence = 100 }
	}
	if = {
		limit = { check_variable = { dd_chi_confidence < 0 } }
		set_variable = { dd_chi_confidence = 0 }
	}
	if = {
		limit = { check_variable = { dd_chi_confidence > 100 } }
		set_variable = { dd_chi_confidence = 100 }
	}
	set_variable = { dd_chi_conf_missing = 100 }
	subtract_variable = { dd_chi_conf_missing = dd_chi_confidence }
	set_variable = { dd_chi_conf_plan = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_plan = -0.002 }
	set_variable = { dd_chi_conf_atk = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_atk = -0.001 }
	set_variable = { dd_chi_conf_org = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_org = -0.001 }
	set_variable = { dd_chi_conf_mor = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_mor = -0.001 }
	set_variable = { dd_chi_conf_cp = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_cp = -0.002 }
	set_variable = { dd_chi_conf_xp = dd_chi_conf_missing }
	multiply_variable = { dd_chi_conf_xp = -0.002 }
	set_variable = { dd_chi_conf_frame = dd_chi_confidence }
	multiply_variable = { dd_chi_conf_frame = 0.19 }
	round_variable = dd_chi_conf_frame
	add_to_variable = { dd_chi_conf_frame = 1 }
	if = {
		limit = { check_variable = { dd_chi_conf_frame < 1 } }
		set_variable = { dd_chi_conf_frame = 1 }
	}
	if = {
		limit = { check_variable = { dd_chi_conf_frame > 20 } }
		set_variable = { dd_chi_conf_frame = 20 }
	}
	if = {
		limit = { has_idea = dd_chi_army_fear }
		remove_ideas = dd_chi_army_fear
	}
	if = {
		limit = { NOT = { original_tag = CHI } }
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_chi_confidence_fear } }
			remove_dynamic_modifier = { modifier = dd_chi_confidence_fear }
		}
	}
	else_if = {

# First war between China and Taiwan. 180 days, then a ceasefire.
# Occupied land returns. A later war does not start another clock.
dd_chi_begin_strait_clock = {
	if = {
		limit = { NOT = { has_global_flag = dd_chi_taiwan_clock_started } }
		set_global_flag = dd_chi_taiwan_clock_started
		CHI = {
			set_country_flag = dd_chi_taiwan_clock_running
			activate_mission = dd_chi_taiwan_countdown
			country_event = { id = dd_chi_tw.1 }
		}
		FOR = {
			set_country_flag = dd_chi_taiwan_clock_running
			activate_mission = dd_chi_taiwan_countdown
			country_event = { id = dd_chi_tw.1 }
		}
	}
}

# One writer. Both mission timeouts can call this on the same day.
dd_chi_strait_ceasefire = {
	if = {
		limit = { NOT = { has_global_flag = dd_chi_taiwan_clock_done } }
		set_global_flag = dd_chi_taiwan_clock_done
		CHI = {
			every_other_country = {
				limit = {
					has_war_with = CHI
					OR = {
						tag = FOR
						has_war_together_with = FOR
					}
				}
				CHI = { white_peace = PREV }
			}
			clr_country_flag = dd_chi_taiwan_clock_running
			country_event = { id = dd_chi_tw.2 }
			if = {
				limit = { NOT = { owns_state = 524 } }
				set_country_flag = dd_chi_taiwan_failed
			}
			dd_chi_regime_watch = yes
		}
		FOR = {
			clr_country_flag = dd_chi_taiwan_clock_running
			country_event = { id = dd_chi_tw.2 }
		}
	}
}

# China's tree, plus the party balance the tree shifts. Not the warlords.
# copy_completed_from marks Beijing's finished focuses done. It does not pay their rewards again.
dd_chi_adopt_tree = {
	if = {
		limit = {
			OR = {
				NOT = { has_focus_tree = dd_chi_focus }
				has_country_flag = dd_chi_sync_focus_progress
			}
		}
		load_focus_tree = {
			tree = dd_chi_focus
			keep_completed = yes
			copy_completed_from = CHI
		}
		clr_country_flag = dd_chi_sync_focus_progress
		if = {
			limit = { has_global_flag = dd_chi_civil_war_started }
			set_country_flag = dd_chi_focus_progress_copied
		}
	}
	if = {
		limit = { NOT = { has_power_balance = { id = CHI_party_balance } } }
		set_power_balance = {
			id = CHI_party_balance
			left_side = CHI_bop_loyalist
			right_side = CHI_bop_technocrat
			set_value = -0.20
		}
	}
	dd_chi_drop_paranoia = yes
	dd_chi_recruit_advisors = yes
}
