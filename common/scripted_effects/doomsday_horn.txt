# Tigray rising, 1 Jan 2026. The TDF never disarmed. The OLA is the militia
# that fought beside them in 2021. Fano is a separate Amhara war.
dd_eth_tigray_rising = {
	if = {
		limit = {
			NOT = { has_country_flag = dd_eth_rebellion_done }
			owns_state = 842
		}
		set_country_flag = dd_eth_rebellion_done
		# Fragile Peace cancels limited conscription, so the pool is empty and a split
		# deletes the field army. Keep every division. Only state 842 changes hands.
		add_manpower = 80000
		if = {
			limit = { NOT = { has_idea = dd_eth_mobilized } }
			add_ideas = dd_eth_mobilized
		}
		start_civil_war = {
			ideology = communism
			size = 0.15
			army_ratio = 0
			navy_ratio = 0
			air_ratio = 0
			capital = 842
			states = { 842 }
			keep_unit_leaders = no
			set_country_flag = dd_eth_tig_side
			save_event_target_as = dd_tig_rebels
		}
		if = {
			limit = { has_event_target = dd_tig_rebels }
			event_target:dd_tig_rebels = {
				set_cosmetic_tag = TIG_REVOLT
				set_country_flag = dd_eth_tig_side
				add_state_core = 842
				set_politics = {
					ruling_party = communism
					elections_allowed = no
				}
				add_popularity = { ideology = communism popularity = 0.6 }
				create_country_leader = {
					name = "Debretsion Gebremichael"
					desc = "TPLF chairman. His wing and the Tigray Defense Forces pushed the federal interim administration out of Mekelle in March 2025."
					picture = GFX_portrait_TIG_debretsion
					ideology = marxism
				}
				# Ethiopia keeps Fragile Peace and Berbera Access. Tigray does not.
				if = {
					limit = { has_idea = dd_fragile_peace }
					remove_ideas = dd_fragile_peace
				}
				if = {
					limit = { has_idea = dd_berbera_access }
					remove_ideas = dd_berbera_access
				}
				add_ideas = dd_tdf_and_ola
				add_manpower = 60000
				add_equipment_to_stockpile = { type = infantry_equipment_cw1 amount = 4000 producer = ETH }
				add_equipment_to_stockpile = { type = support_equipment_cw1 amount = 80 producer = ETH }
				add_equipment_to_stockpile = { type = artillery_equipment amount = 24 producer = ETH }
				if = {
					limit = { ROOT = { has_country_flag = dd_eth_tdf_armed } }
					add_manpower = 20000
					add_equipment_to_stockpile = { type = infantry_equipment_cw1 amount = 2000 producer = ETH }
				}
				load_oob = "TIG_revolt"
				load_focus_tree = { tree = dd_tig_focus keep_completed = no }
			}
		}
	}
}

# Player completed Prop Up Tigray. Ethiopia keeps the war. The player takes Mekelle.
dd_eth_player_becomes_tigray = {
	if = {
		limit = {
			has_country_flag = dd_eth_player_tigray
			NOT = { has_country_flag = dd_eth_tig_side }
			any_country = { has_country_flag = dd_eth_tig_side }
		}
		if = {
			limit = { has_event_target = dd_tig_rebels }
			event_target:dd_tig_rebels = { change_tag_from = ROOT }
		}
		else = {
			every_country = {
				limit = { has_country_flag = dd_eth_tig_side }
				change_tag_from = ROOT
			}
		}
	}
}

dd_uae_open_compact = {
	if = {
		limit = { is_in_faction = no }
		hidden_effect = {
			set_rule = { can_create_factions = yes }
		}
		create_faction = DD_FACTION_RED_SEA
	}
}

# Called on the country being invited. ROOT is the Emirates.
# STC must be at peace. Somaliland must still hold and be at peace.
# Only the Rapid Support Forces come out of the Sudan war. Sudan and Eritrea never join.
dd_uae_try_invite = {
	if = {
		limit = {
			ROOT = { is_faction_leader = yes }
			exists = yes
			is_subject = no
			is_in_faction = no
			NOT = { has_war_with = ROOT }
			OR = {
				AND = { tag = STC has_war = no }
				AND = { tag = SML has_war = no owns_state = 269 }
				AND = {
					tag = RSF
					dd_sudan_war_over = yes
				}
			}
		}
		ROOT = { add_to_faction = PREV }
	}
}

# Marks the guarantee round. Does not found a faction.
dd_levant_open = {
	set_country_flag = dd_levant_guarantees
}

# Scope is the partner. ROOT is the country taking the focus.
# Each guarantees the other. Existing factions are left alone.
dd_levant_invite = {
	if = {
		limit = {
			exists = yes
			is_subject = no
			NOT = { tag = ROOT }
			NOT = { has_war_with = ROOT }
		}
		if = {
			limit = { NOT = { has_guaranteed = ROOT } }
			give_guarantee = ROOT
		}
		ROOT = {
			if = {
				limit = { NOT = { has_guaranteed = PREV } }
				give_guarantee = PREV
			}
		}
	}
}
