# Hormuz map. Naval zones follow real naval control. Bomb spirits follow a raid on the same day.
# Arabian Sea is strategic region 104. Persian Gulf is strategic region 203.
# The strait follows whoever holds the Persian Gulf.

# Saudi Arabia and Oman stay out of the war. The squeeze still damages Saudi buildings through the idea.
dd_hormuz_join_gulf = {
	every_country = {
		limit = {
			exists = yes
			NOT = { has_war_with = PER }
			OR = {
				tag = KUW
				tag = QAT
				tag = UAE
				tag = BHR
			}
		}
		add_to_war = {
			targeted_alliance = USA
			enemy = PER
		}
	}
	USA = { dd_usa_sync_partner_air = yes }
}

# One Gulf tag. The caller clears dd_hormuz_gulf_bombed first, then this sets it again if a state was bombed today.
dd_hormuz_sync_gulf_spirit = {
	if = {
		limit = {
			any_owned_state = { days_since_last_strategic_bombing < 1 }
		}
		set_global_flag = dd_hormuz_gulf_bombed
		if = {
			limit = { NOT = { has_idea = dd_hormuz_gulf_bombed } }
			add_ideas = dd_hormuz_gulf_bombed
		}
	}
	else_if = {
		limit = { has_idea = dd_hormuz_gulf_bombed }
		remove_ideas = dd_hormuz_gulf_bombed
	}
}

# One card toward collapse. Used when a quarter of bombing clears the 7-day bar.
dd_hormuz_auth_worse = {
	if = {
		limit = { PER = { check_variable = { dd_hormuz_auth < 5 } } }
		PER = { add_to_variable = { dd_hormuz_auth = 1 } }
	}
	dd_hormuz_apply_authority = yes
}

# One card toward Total Control. Rally, Total War, and Ration Basic Goods.
dd_hormuz_auth_better = {
	if = {
		limit = {
			has_global_flag = dd_hormuz_war
			PER = { check_variable = { dd_hormuz_auth > 1 } }
		}
		PER = { subtract_from_variable = { dd_hormuz_auth = 1 } }
		dd_hormuz_apply_authority = yes
	}
}

# The bomb-day mark is counted when the raid is seen. This only advances the 90-day quarter.
dd_hormuz_auth_daily = {
	add_to_variable = { global.dd_hormuz_quarter_elapsed = 1 }
	if = {
		limit = { check_variable = { global.dd_hormuz_quarter_elapsed > 89 } }
		if = {
			limit = { check_variable = { global.dd_hormuz_quarter_bomb_days > 6 } }
			dd_hormuz_auth_worse = yes
		}
		set_variable = { global.dd_hormuz_quarter_elapsed = 0 }
		set_variable = { global.dd_hormuz_quarter_bomb_days = 0 }
	}
}

# Bomb spirits only. One pass, from the United States, during the war.
# Ideas change when the result changes, so an hour with no new raid does not touch them.
dd_hormuz_sync_bombs = {
	if = {
		limit = { dd_hormuz_per_bombed_now = yes }
		if = {
			limit = { NOT = { has_global_flag = dd_hormuz_per_bombed } }
			set_global_flag = dd_hormuz_per_bombed
			PER = {
				add_ideas = dd_hormuz_per_bombed
				set_country_flag = dd_per_bomb_cash_month
			}
		}
		# A raid refreshes a five-day missile salvo. Quiet days let it run out.
		PER = {
			if = {
				limit = { has_idea = dd_per_total_war }
				set_country_flag = {
					flag = dd_per_missile_salvo
					days = 5
					value = 1
				}
			}
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = global.dd_hormuz_per_bomb_day
						value = global.num_days
						compare = equals
					}
				}
			}
			set_variable = { global.dd_hormuz_per_bomb_day = global.num_days }
			add_to_variable = { global.dd_hormuz_quarter_bomb_days = 1 }
		}
	}
	else_if = {
		limit = { has_global_flag = dd_hormuz_per_bombed }
		clr_global_flag = dd_hormuz_per_bombed
		PER = { remove_ideas = dd_hormuz_per_bombed }
	}

	clr_global_flag = dd_hormuz_gulf_bombed
	if = {
		limit = { SAU = { exists = yes } }
		SAU = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { KUW = { exists = yes } }
		KUW = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { QAT = { exists = yes } }
		QAT = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { UAE = { exists = yes } }
		UAE = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { BHR = { exists = yes } }
		BHR = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { OMA = { exists = yes } }
		OMA = { dd_hormuz_sync_gulf_spirit = yes }
	}
	if = {
		limit = { has_global_flag = dd_hormuz_gulf_bombed }
		set_variable = { global.dd_hormuz_gulf_bomb_day = global.num_days }
	}

	if = {
		limit = { dd_hormuz_gulf_cities_bombed_now = yes }
		if = {
			limit = { NOT = { has_global_flag = dd_hormuz_gulf_cities_bombed } }
			set_global_flag = dd_hormuz_gulf_cities_bombed
			if = {
				limit = { USA = { NOT = { has_idea = dd_hormuz_gulfbomb_usa } } }
				USA = { add_ideas = dd_hormuz_gulfbomb_usa }
			}
			dd_hormuz_blockade_targets = yes
		}
	}
	else_if = {
		limit = { has_global_flag = dd_hormuz_gulf_cities_bombed }
		clr_global_flag = dd_hormuz_gulf_cities_bombed
		if = {
			limit = { USA = { has_idea = dd_hormuz_gulfbomb_usa } }
			USA = { remove_ideas = dd_hormuz_gulfbomb_usa }
		}
		dd_hormuz_blockade_targets = yes
	}

	if = {
		limit = { check_variable = { global.dd_hormuz_per_bomb_day > 0 } }
		set_variable = { global.dd_hormuz_per_bomb_ago = global.num_days }
		subtract_from_variable = { global.dd_hormuz_per_bomb_ago = global.dd_hormuz_per_bomb_day }
	}
	if = {
		limit = { check_variable = { global.dd_hormuz_gulf_bomb_day > 0 } }
		set_variable = { global.dd_hormuz_gulf_bomb_ago = global.num_days }
		subtract_from_variable = { global.dd_hormuz_gulf_bomb_ago = global.dd_hormuz_gulf_bomb_day }
	}
}
