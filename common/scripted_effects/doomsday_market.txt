# Fixed prices. Host and clients must never disagree.

dd_buy_infantry = {
	if = {
		limit = { check_variable = { treasury > 9.99 } }
		subtract_from_variable = { treasury = 10 }
		add_equipment_to_stockpile = {
			type = infantry_equipment_3
			amount = 100
			producer = ROOT
		}
	}
}

dd_sell_infantry = {
	if = {
		limit = { has_equipment = { infantry_equipment > 99 } }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = -100
		}
		add_to_variable = { treasury = 5 }
	}
}
