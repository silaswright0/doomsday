dd_isr_hold_election = {
	if = {
		limit = { NOT = { has_country_flag = dd_isr_election_done } }
		set_country_flag = dd_isr_election_done
		set_variable = { dd_isr_hadash = party_popularity@progressive_populism }
		add_to_variable = { dd_isr_hadash = party_popularity@communism }
		set_variable = { dd_isr_dem = party_popularity@social_democracy }
		set_variable = { dd_isr_yashar = party_popularity@social_liberalism }
		set_variable = { dd_isr_bibi = party_popularity@liberal_conservatism }
		add_to_variable = { dd_isr_bibi = party_popularity@national_populism }
		add_to_variable = { dd_isr_bibi = party_popularity@theocratic_absolutism }
		add_to_variable = { dd_isr_bibi = party_popularity@fascism }
		set_country_flag = dd_isr_resolving_election
		if = {
			limit = {
				check_variable = { dd_isr_hadash > dd_isr_dem }
				check_variable = { dd_isr_hadash > dd_isr_yashar }
				check_variable = { dd_isr_hadash > dd_isr_bibi }
			}
			complete_national_focus = ISR_election_hadash
			news_event = { id = dd_isr_news.1 }
		}
		else_if = {
			limit = {
				check_variable = { dd_isr_dem > dd_isr_yashar }
				check_variable = { dd_isr_dem > dd_isr_bibi }
			}
			complete_national_focus = ISR_election_democrats
			news_event = { id = dd_isr_news.2 }
		}
		else_if = {
			limit = { check_variable = { dd_isr_yashar > dd_isr_bibi } }
			complete_national_focus = ISR_election_yashar
			news_event = { id = dd_isr_news.3 }
		}
		else = {
			complete_national_focus = ISR_election_netanyahu
			news_event = { id = dd_isr_news.4 }
		}
		clr_country_flag = dd_isr_resolving_election
	}
}

# The IDF card stays. Urban fighting and combined arms rewrite its numbers.
dd_isr_refresh_idf = {
	if = {
		limit = {
			OR = {
				has_idea = dd_isr_idf_urban
				has_idea = dd_isr_idf_both
			}
		}
		set_country_flag = dd_isr_idf_urban
	}
	if = {
		limit = {
			OR = {
				has_idea = dd_isr_idf_arms
				has_idea = dd_isr_idf_both
			}
		}
		set_country_flag = dd_isr_idf_arms
	}
	set_variable = { dd_idf_xp = 0.20 }
	set_variable = { dd_idf_plan = 0.20 }
	set_variable = { dd_idf_morale = 0.15 }
	set_variable = { dd_idf_move = -0.20 }
	set_variable = { dd_idf_org = 0.10 }
	set_variable = { dd_idf_atk = 0 }
	set_variable = { dd_idf_def = 0 }
	set_variable = { dd_idf_spd = 0 }
	set_variable = { dd_idf_arm = 0 }
	set_variable = { dd_idf_brk = 0 }
	if = {
		limit = { has_country_flag = dd_isr_idf_urban }
		set_variable = { dd_idf_org = 0.15 }
		set_variable = { dd_idf_atk = 0.10 }
		set_variable = { dd_idf_def = 0.10 }
	}
	if = {
		limit = { has_country_flag = dd_isr_idf_arms }
		if = {
			limit = { has_country_flag = dd_isr_idf_urban }
			set_variable = { dd_idf_org = 0.20 }
			set_variable = { dd_idf_atk = 0.20 }
		}
		else = {
			set_variable = { dd_idf_org = 0.15 }
			set_variable = { dd_idf_atk = 0.10 }
		}
		set_variable = { dd_idf_spd = 0.15 }
		set_variable = { dd_idf_arm = 0.10 }
		set_variable = { dd_idf_brk = 0.10 }
	}
	if = { limit = { has_idea = dd_isr_idf } remove_ideas = dd_isr_idf }
	if = { limit = { has_idea = dd_isr_idf_urban } remove_ideas = dd_isr_idf_urban }
	if = { limit = { has_idea = dd_isr_idf_arms } remove_ideas = dd_isr_idf_arms }
	if = { limit = { has_idea = dd_isr_idf_both } remove_ideas = dd_isr_idf_both }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = dd_isr_idf_spirit } } }
		add_dynamic_modifier = { modifier = dd_isr_idf_spirit }
	}
	force_update_dynamic_modifier = yes
}
