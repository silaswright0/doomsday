dd_isr_hold_election = {
	if = {
		limit = { NOT = { has_country_flag = dd_isr_election_done } }
		set_country_flag = dd_isr_election_done
		set_variable = { dd_isr_hadash = party_popularity@progressive_populism }
		add_to_variable = { dd_isr_hadash = party_popularity@communism }
		set_variable = { dd_isr_dem = party_popularity@social_democracy }
		set_variable = { dd_isr_yashar = party_popularity@social_liberalism }
		set_variable = { dd_isr_bibi = party_popularity@liberal_conservatism }
		add_to_variable = { dd_isr_bibi = party_popularity@national_populism }
		add_to_variable = { dd_isr_bibi = party_popularity@theocratic_absolutism }
		add_to_variable = { dd_isr_bibi = party_popularity@fascism }
		set_country_flag = dd_isr_resolving_election
		if = {
			limit = {
				check_variable = { dd_isr_hadash > dd_isr_dem }
				check_variable = { dd_isr_hadash > dd_isr_yashar }
				check_variable = { dd_isr_hadash > dd_isr_bibi }
			}
			complete_national_focus = ISR_election_hadash
			news_event = { id = dd_isr_news.1 }
		}
		else_if = {
			limit = {
				check_variable = { dd_isr_dem > dd_isr_yashar }
				check_variable = { dd_isr_dem > dd_isr_bibi }
			}
			complete_national_focus = ISR_election_democrats
			news_event = { id = dd_isr_news.2 }
		}
		else_if = {
			limit = { check_variable = { dd_isr_yashar > dd_isr_bibi } }
			complete_national_focus = ISR_election_yashar
			news_event = { id = dd_isr_news.3 }
		}
		else = {
			complete_national_focus = ISR_election_netanyahu
			news_event = { id = dd_isr_news.4 }
		}
		clr_country_flag = dd_isr_resolving_election
	}
}

dd_isr_palestine_war = {
	custom_effect_tooltip = dd_isr_palestine_war_tt
	hidden_effect = {
		set_cosmetic_tag = ISR_PALESTINE
		set_country_flag = dd_isr_palestine_declared
		if = {
			limit = {
				1120 = { is_owned_by = ROOT }
				454 = { is_owned_by = ROOT }
				1088 = { is_owned_by = ROOT }
			}
			set_capital = { state = 1120 }
			start_civil_war = {
				ideology = theocratic_absolutism
				size = 0.35
				capital = 454
				states = { 454 1088 }
				set_country_flag = dd_isr_revolt
				save_event_target_as = dd_isr_rebels
			}
		}
		else_if = {
			limit = {
				1120 = { is_owned_by = ROOT }
				454 = { is_owned_by = ROOT }
			}
			set_capital = { state = 1120 }
			start_civil_war = {
				ideology = theocratic_absolutism
				size = 0.35
				capital = 454
				states = { 454 }
				set_country_flag = dd_isr_revolt
				save_event_target_as = dd_isr_rebels
			}
		}
		if = {
			limit = { has_event_target = dd_isr_rebels }
			event_target:dd_isr_rebels = {
				set_cosmetic_tag = ISR_REVOLT
				set_country_flag = dd_isr_revolt
				set_politics = { ruling_party = theocratic_absolutism elections_allowed = no }
				add_popularity = { ideology = theocratic_absolutism popularity = 0.45 }
				add_popularity = { ideology = fascism popularity = 0.35 }
				create_country_leader = {
					name = "Itamar Ben-Gvir"
					ideology = theocrat
				}
				add_ideas = dd_isr_insurgent
			}
		}
		news_event = { id = dd_isr_news.5 }
	}
}
