# Monthly. Economic strain feeds communism. A stalled year, lost cores, and lost battles feed fascism.
dd_sov_anger_month = {
	if = {
		limit = { has_war_with = UKR }
		if = {
			limit = {
				check_variable = { dd_sov_ukr_stage > dd_sov_stage_before }
				check_variable = { dd_sov_stage_before > -0.1 }
			}
			add_popularity = { ideology = fascism popularity = 0.03 }
		}
		if = {
			limit = { has_variable = dd_sov_ukr_cg }
			set_variable = { dd_sov_red_gain = dd_sov_ukr_cg }
			multiply_variable = { dd_sov_red_gain = 0.02 }
			add_popularity = { ideology = communism popularity = var:dd_sov_red_gain }
			if = {
				limit = { check_variable = { dd_sov_ukr_cg > 0.30 } }
				set_variable = { dd_sov_red_direct = dd_sov_ukr_cg }
				subtract_from_variable = { dd_sov_red_direct = 0.30 }
				add_popularity = { ideology = communism popularity = var:dd_sov_red_direct }
			}
		}
		set_variable = { dd_sov_cores_now = 0 }
		every_core_state = {
			limit = { is_owned_by = ROOT }
			add_to_variable = { ROOT.dd_sov_cores_now = 1 }
		}
		if = {
			limit = { NOT = { has_country_flag = dd_sov_cores_ready } }
			set_variable = { dd_sov_cores_last = dd_sov_cores_now }
			set_country_flag = dd_sov_cores_ready
		}
		else = {
			if = {
				limit = { check_variable = { dd_sov_cores_last > dd_sov_cores_now } }
				set_variable = { dd_sov_cores_lost = dd_sov_cores_last }
				subtract_from_variable = { dd_sov_cores_lost = dd_sov_cores_now }
				multiply_variable = { dd_sov_cores_lost = 0.05 }
				add_popularity = { ideology = fascism popularity = var:dd_sov_cores_lost }
			}
			set_variable = { dd_sov_cores_last = dd_sov_cores_now }
		}
		if = {
			limit = { NOT = { has_variable = dd_sov_battles_lost } }
			set_variable = { dd_sov_battles_lost = 0 }
		}
		if = {
			limit = { NOT = { has_variable = dd_sov_lost_last } }
			set_variable = { dd_sov_lost_last = 0 }
		}
		set_variable = { dd_sov_lost_month = dd_sov_battles_lost }
		subtract_from_variable = { dd_sov_lost_month = dd_sov_lost_last }
		set_variable = { dd_sov_lost_last = dd_sov_battles_lost }
		if = {
			limit = { check_variable = { dd_sov_lost_month > 0 } }
			set_variable = { dd_sov_battle_gain = dd_sov_lost_month }
			multiply_variable = { dd_sov_battle_gain = 0.0001 }
			add_popularity = { ideology = fascism popularity = var:dd_sov_battle_gain }
		}
	}
}
