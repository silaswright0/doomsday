# Ukraine war clock. Ten years from 24 February 2022 to February 2032.
# 2026.1.1 is month 46 of 120, year 4 of 10 (The Grind, 2025).
# One stage per year: 0, 12, 24, 36, 48, 60, 72, 84, 96, 108.
# A peace freezes the month count. The spirit comes off until the war resumes.

	remove_ideas = dd_sov_ukr_2

	remove_ideas = dd_sov_ukr_5

	remove_ideas = dd_sov_ukr_8
	remove_ideas = dd_sov_ukr_9
	set_variable = { dd_sov_ukr_stage_applied = -1 }
}

dd_sov_ukr_sync = {
	if = {
		limit = { check_variable = { dd_sov_ukr_months < 12 } }
		set_variable = { dd_sov_ukr_stage = 0 }
		set_variable = { dd_sov_ukr_next_at = 12 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 24 } }
		set_variable = { dd_sov_ukr_stage = 1 }
		set_variable = { dd_sov_ukr_next_at = 24 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 36 } }
		set_variable = { dd_sov_ukr_stage = 2 }
		set_variable = { dd_sov_ukr_next_at = 36 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 48 } }
		set_variable = { dd_sov_ukr_stage = 3 }
		set_variable = { dd_sov_ukr_next_at = 48 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 60 } }
		set_variable = { dd_sov_ukr_stage = 4 }
		set_variable = { dd_sov_ukr_next_at = 60 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 72 } }
		set_variable = { dd_sov_ukr_stage = 5 }
		set_variable = { dd_sov_ukr_next_at = 72 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 84 } }
		set_variable = { dd_sov_ukr_stage = 6 }
		set_variable = { dd_sov_ukr_next_at = 84 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 96 } }
		set_variable = { dd_sov_ukr_stage = 7 }
		set_variable = { dd_sov_ukr_next_at = 96 }
	}
	else_if = {
		limit = { check_variable = { dd_sov_ukr_months < 108 } }
		set_variable = { dd_sov_ukr_stage = 8 }
		set_variable = { dd_sov_ukr_next_at = 108 }
	}
	else = {
		set_variable = { dd_sov_ukr_stage = 9 }
		set_variable = { dd_sov_ukr_next_at = dd_sov_ukr_months }
	}
	set_variable = { dd_sov_ukr_year = dd_sov_ukr_stage }
	add_to_variable = { dd_sov_ukr_year = 1 }
	set_variable = { dd_sov_ukr_until = dd_sov_ukr_next_at }
	subtract_from_variable = { dd_sov_ukr_until = dd_sov_ukr_months }
	if = {
		limit = { check_variable = { dd_sov_ukr_until < 0 } }
		set_variable = { dd_sov_ukr_until = 0 }
	}
	if = {
		limit = {
			NOT = { check_variable = { dd_sov_ukr_stage_applied = dd_sov_ukr_stage } }
		}
		dd_sov_ukr_apply = yes
	}
}

dd_sov_ukr_monthly = {
	if = {
		limit = { has_war_with = UKR }
		if = {
			limit = { NOT = { has_country_flag = dd_sov_ukr_clock } }
			set_variable = { dd_sov_ukr_months = 0 }
			set_variable = { dd_sov_ukr_stage_applied = -1 }
			set_country_flag = dd_sov_ukr_clock
		}
		else = {
			add_to_variable = { dd_sov_ukr_months = 1 }
		}
		dd_sov_ukr_sync = yes
	}
	else = {
		dd_sov_ukr_clear = yes
	}
}
