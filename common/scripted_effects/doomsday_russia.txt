# February 2032. Ukrainian cores come home. Russia is not annexed.
dd_sov_ukr_surrender = {
	if = {
		limit = {
			has_war_with = UKR
			NOT = { has_country_flag = dd_sov_ukr_surrendered }
		}
		set_country_flag = dd_sov_ukr_surrendered
		set_country_flag = dd_sov_ukr_first_war_over
		every_owned_state = {
			limit = { is_core_of = UKR }
			transfer_state_to = UKR
		}
		every_country = {
			limit = {
				OR = {
					tag = ROOT
					is_in_faction_with = ROOT
				}
			}
			every_country = {
				limit = {
					has_war_with = PREV
					OR = {
						tag = UKR
						is_in_faction_with = UKR
					}
				}
				white_peace = PREV
			}
		}
		dd_sov_ukr_clear = yes
		country_event = { id = dd_sov_ukr.1 }
		UKR = { country_event = { id = dd_sov_ukr.1 } }
	}
}

		set_variable = { dd_sov_ukr_stage = 0 }

		limit = { check_variable = { dd_sov_ukr_months < 24 } }
		set_variable = { dd_sov_ukr_stage = 1 }
		set_variable = { dd_sov_ukr_next_at = 24 }

	else_if = {

		set_variable = { dd_sov_ukr_stage = 2 }
		set_variable = { dd_sov_ukr_next_at = 36 }

	else_if = {

		set_variable = { dd_sov_ukr_stage = 4 }
		set_variable = { dd_sov_ukr_next_at = 60 }

		set_variable = { dd_sov_ukr_stage = 5 }

		set_variable = { dd_sov_ukr_next_at = 84 }
	}

		limit = { check_variable = { dd_sov_ukr_months < 96 } }
		set_variable = { dd_sov_ukr_stage = 7 }

		set_variable = { dd_sov_ukr_stage = 9 }

	}
	else = {

		limit = { NOT = { has_country_flag = dd_sov_ukr_first_war_over } }

dd_sov_fuel_sync = {
	if = {
		limit = { NOT = { has_variable = dd_sov_fuel_cap } }
		set_variable = { dd_sov_fuel_cap = 20 }
	}
	if = {
		limit = { check_variable = { dd_sov_fuel > dd_sov_fuel_cap } }
		set_variable = { dd_sov_fuel = dd_sov_fuel_cap }
	}
	if = {
		limit = { check_variable = { dd_sov_fuel < 0 } }
		set_variable = { dd_sov_fuel = 0 }
	}
	set_variable = { dd_sov_fuel_gain = dd_sov_fuel }
	multiply_variable = { dd_sov_fuel_gain = -0.04 }
	set_variable = { dd_sov_fuel_use = dd_sov_fuel }
	multiply_variable = { dd_sov_fuel_use = 0.03 }
	set_variable = { dd_sov_fuel_fac = dd_sov_fuel }
	multiply_variable = { dd_sov_fuel_fac = -0.02 }
	if = {
		limit = { check_variable = { dd_sov_fuel > 0 } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = dd_sov_fuel_fires } } }
			add_dynamic_modifier = { modifier = dd_sov_fuel_fires }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_sov_fuel_fires } }
			remove_dynamic_modifier = { modifier = dd_sov_fuel_fires }
		}
	}
}

dd_sov_fuel_apply_hit = {

	}
}
