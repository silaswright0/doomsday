# Repair +30% is on the idea. Iran Against the World swaps in the core and factory card.
dd_per_apply_final_battle = {
	if = {
		limit = {
			OR = {
				has_idea = dd_per_final_battle_2
				has_completed_focus = PER_iran_against_the_world
			}
		}
		set_country_flag = dd_per_final_battle_world
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = dd_per_final_battle_spirit } }
		remove_dynamic_modifier = { modifier = dd_per_final_battle_spirit }
	}
	if = {
		limit = { has_country_flag = dd_per_final_battle_world }
		if = { limit = { has_idea = dd_per_final_battle } remove_ideas = dd_per_final_battle }
		if = { limit = { NOT = { has_idea = dd_per_final_battle_2 } } add_ideas = dd_per_final_battle_2 }
	}
	else = {
		if = { limit = { has_idea = dd_per_final_battle_2 } remove_ideas = dd_per_final_battle_2 }
		if = { limit = { NOT = { has_idea = dd_per_final_battle } } add_ideas = dd_per_final_battle }
	}
}

# Strait and Saudi squeeze. Gulf-city bombing is owned by dd_hormuz_sync_bombs.
# Importers on the list above take the consumer-goods shock. Runs once a day from the USA.
dd_hormuz_econ_sync = {
	if = {
		limit = {
			has_global_flag = dd_hormuz_escort_shipping
			OR = {
				NOT = { USA = { has_naval_control = 104 } }
				NOT = { USA = { has_naval_control = 203 } }
				NOT = { has_global_flag = dd_hormuz_blockade }
			}
		}
		clr_global_flag = dd_hormuz_escort_shipping
	}
	if = {
		limit = { SAU = { has_idea = dd_sau_squeezed } }
		set_global_flag = dd_hormuz_squeeze_on
	}
	else = {
		clr_global_flag = dd_hormuz_squeeze_on
	}
	if = {
		limit = {
			has_global_flag = dd_hormuz_blockade
			has_global_flag = dd_hormuz_escort_shipping
			USA = { exists = yes }
		}
		USA = {
			if = {
				limit = { has_idea = dd_hormuz_blockade_usa }
				remove_ideas = dd_hormuz_blockade_usa
			}
			if = {
				limit = { NOT = { has_idea = dd_hormuz_blockade_usa_half } }
				add_ideas = dd_hormuz_blockade_usa_half
			}
		}
	}
	else_if = {
		limit = {
			has_global_flag = dd_hormuz_blockade
			USA = { exists = yes }
		}
		USA = {
			if = {
				limit = { has_idea = dd_hormuz_blockade_usa_half }
				remove_ideas = dd_hormuz_blockade_usa_half
			}
			if = {
				limit = { NOT = { has_idea = dd_hormuz_blockade_usa } }
				add_ideas = dd_hormuz_blockade_usa
			}
		}
	}
	else = {
		USA = {
			if = {
				limit = { has_idea = dd_hormuz_blockade_usa }
				remove_ideas = dd_hormuz_blockade_usa
			}
			if = {
				limit = { has_idea = dd_hormuz_blockade_usa_half }
				remove_ideas = dd_hormuz_blockade_usa_half
			}
		}
	}
	if = {
		limit = {
			has_global_flag = dd_hormuz_squeeze_on
			USA = { exists = yes }
			NOT = { USA = { has_idea = dd_hormuz_saudi_shock } }
		}
		USA = { add_ideas = dd_hormuz_saudi_shock }
	}
	else_if = {
		limit = {
			NOT = { has_global_flag = dd_hormuz_squeeze_on }
			USA = { has_idea = dd_hormuz_saudi_shock }
		}
		USA = { remove_ideas = dd_hormuz_saudi_shock }
	}
	if = {
		limit = {
			has_global_flag = dd_hormuz_gulf_cities_bombed
			USA = { exists = yes }
			NOT = { USA = { has_idea = dd_hormuz_gulfbomb_usa } }
		}
		USA = { add_ideas = dd_hormuz_gulfbomb_usa }
	}
	else_if = {
		limit = {
			NOT = { has_global_flag = dd_hormuz_gulf_cities_bombed }
			USA = { has_idea = dd_hormuz_gulfbomb_usa }
		}
		USA = { remove_ideas = dd_hormuz_gulfbomb_usa }
	}
	dd_hormuz_blockade_targets = yes
}

# Oil Revenue pays nothing while either strait spirit is on. The idea comes back when both are gone.
dd_hormuz_suspend_oil_revenue = {
	if = {
		limit = {
			has_idea = dd_oil_revenue
			OR = {
				has_idea = dd_gulf_oil_sealed
				has_idea = dd_per_oil_blocked
			}
		}
		set_country_flag = dd_oil_revenue_suspended
		remove_ideas = dd_oil_revenue
	}
}

dd_hormuz_restore_oil_revenue = {
	if = {
		limit = {
			has_country_flag = dd_oil_revenue_suspended
			NOT = { has_idea = dd_gulf_oil_sealed }
			NOT = { has_idea = dd_per_oil_blocked }
		}
		clr_country_flag = dd_oil_revenue_suspended
		add_ideas = dd_oil_revenue
	}
}

# Iran only. Called after the sea flags are current, and again when the blockade starts.
dd_hormuz_sync_per_oil = {
	if = {
		limit = { PER = { has_idea = dd_gulf_oil_sealed } }
		PER = { remove_ideas = dd_gulf_oil_sealed }
	}
	if = {
		limit = { dd_hormuz_per_oil_blocked = yes }
		if = {
			limit = { PER = { NOT = { has_idea = dd_per_oil_blocked } } }
			PER = { add_ideas = dd_per_oil_blocked }
		}
		PER = {
			set_country_flag = dd_per_oil_cash_month
			dd_hormuz_suspend_oil_revenue = yes
		}
	}
	else_if = {
		limit = { PER = { has_idea = dd_per_oil_blocked } }
		PER = {
			remove_ideas = dd_per_oil_blocked
			dd_hormuz_restore_oil_revenue = yes
		}
	}
	else = {
		PER = { dd_hormuz_restore_oil_revenue = yes }
	}
}

				}
				retire_country_leader = yes
				create_country_leader = {
					name = "Falih al-Fayyad"
					desc = "Chairman of the Popular Mobilization Forces."
					picture = GFX_portrait_IRQ_fayyad
					expire = "2065.1.1"
					ideology = theocrat
				}
				set_technology = { militia_tech = 1 }
				set_variable = { dd_cw_militia_count = 6 }
				dd_per_cw_arm = yes
			}
		}
		if = {
			limit = { NOT = { event_target:dd_shiite_side = { is_subject_of = ROOT } } }
			puppet = event_target:dd_shiite_side
		}
		event_target:dd_shiite_side = {
			if = {
				limit = {
					IRQ = { exists = yes }
					NOT = { has_war_with = IRQ }
				}
				declare_war_on = {
					target = IRQ
					type = dd_annex
				}
			}
		}
		if = {
			limit = {
				IRQ = { exists = yes }
				event_target:dd_shiite_side = { has_war_with = IRQ }
				NOT = { has_war_together_with = event_target:dd_shiite_side }
			}

				targeted_alliance = event_target:dd_shiite_side
				enemy = IRQ
			}
		}
	}
}
