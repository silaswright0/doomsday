# Iranian state collapse. PER becomes the IRGC. The revolts take their regions.
# IRGC, monarchists, and progressive populists fight every released faction.
# Kurds, Balochs, Ahwazi, and Azeris fight only those three.
# No random. The popup only chooses which tag the human keeps playing.

dd_per_cw_arm = {
	set_technology = { militia_tech = 1 }
	division_template = {
		name = "Militia"
		is_locked = yes
		regiments = {
			militia = { x = 0 y = 0 }
			militia = { x = 0 y = 1 }
			militia = { x = 0 y = 2 }
			militia = { x = 1 y = 0 }
			militia = { x = 1 y = 1 }
			militia = { x = 1 y = 2 }
		}
	}
	while_loop_effect = {
		limit = { check_variable = { dd_cw_militia_count > 0 } }
		capital_scope = {
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.6"
				owner = PREV
				count = 1
			}
		}
		subtract_variable = { dd_cw_militia_count = 1 }
	}
}

		}

			limit = {
				OR = {
					KRD = { owns_state = 1001 }

				}
			}
			KRD = {
				set_country_flag = dd_per_cw_released

				dd_per_cw_arm = yes

		}

			}
			PER = { add_civil_war_target = AHZ }
		}

		if = {
			limit = { 1000 = { is_owned_by = PER } }
			AZI = {
				transfer_state = 1000
				add_state_core = 1000
			}
		}
		if = {
			limit = { 419 = { is_owned_by = PER } }
			AZI = {
				transfer_state = 419
				add_state_core = 419
			}
		}
		if = {

		}
		else_if = {
			limit = { AZI = { owns_state = 419 } }
			AZI = { set_capital = { state = 419 } }
		}
		if = {
			limit = {
				OR = {
					AZI = { owns_state = 1000 }
					AZI = { owns_state = 419 }
				}
			}
			AZI = {
				set_country_flag = dd_per_cw_released
				set_variable = { dd_cw_militia_count = 6 }
				dd_per_cw_arm = yes
				declare_war_on = { target = PER type = annex_everything }
				add_civil_war_target = PER

			PER = { add_civil_war_target = AZI }

			limit = { 412 = { is_owned_by = PER } }

				transfer_state = 412
				add_state_core = 412
			}
		}
		if = {
			limit = { 411 = { is_owned_by = PER } }

				transfer_state = 411

			}
		}

dd_per_install_pakpour = {
	if = {
		limit = { NOT = { has_character = PER_pakpour_head } }
		recruit_character = PER_pakpour_head
	}
	if = {
		limit = { NOT = { has_country_leader = { name = "Mohammad Pakpour" ruling_only = yes } } }
		retire_country_leader = yes
		promote_character = PER_pakpour_head
	}
}

dd_pah_install_pahlavi = {
	set_politics = { ruling_party = absolute_monarchy elections_allowed = no }
	if = {
		limit = { NOT = { has_character = PAH_reza_pahlavi } }
		recruit_character = PAH_reza_pahlavi
	}
	if = {
		limit = { NOT = { has_country_leader = { name = "Reza Pahlavi" ruling_only = yes } } }
		retire_country_leader = yes
		promote_character = PAH_reza_pahlavi
	}
}

	}
}
