# Iranian state collapse. PER becomes the IRGC. The revolts take their regions.
# IRGC, monarchists, and progressive populists fight every released faction.
# Kurds, Balochs, Ahwazi, and Azeris fight only those three.
# No random. The popup only chooses which tag the human keeps playing.

# The foreign war ends. The Guard is left to the civil war.
dd_per_foreigners_leave = {
	if = {
		limit = { USA = { exists = yes } }
		USA = { country_event = { id = dd_hormuz.20 } }
	}
	if = {
		limit = { ISR = { exists = yes } }
		ISR = { country_event = { id = dd_hormuz.20 } }
	}
	every_country = {
		limit = {
			OR = {
				tag = PER
				tag = HAM
				tag = HEZ
				tag = HOU
				is_subject_of = PER
			}
		}
		every_enemy_country = {
			limit = {
				NOT = { tag = KRD }
				NOT = { tag = BLS }
				NOT = { tag = AHZ }
				NOT = { tag = AZI }
				NOT = { tag = PAH }
				NOT = { tag = PRP }
			}
			white_peace = PREV
		}
	}
	if = {
		limit = { USA = { has_idea = dd_usa_iran_war } }
		USA = { remove_ideas = dd_usa_iran_war }
	}
	if = {
		limit = { has_global_flag = dd_hormuz_war }
		dd_hormuz_end = yes
	}
}

dd_per_cw_arm = {
	set_technology = { militia_tech = 1 }
	division_template = {
		name = "Militia"
		is_locked = yes
		regiments = {
			militia = { x = 0 y = 0 }
			militia = { x = 0 y = 1 }
			militia = { x = 0 y = 2 }
			militia = { x = 1 y = 0 }
			militia = { x = 1 y = 1 }
			militia = { x = 1 y = 2 }
		}
	}
	while_loop_effect = {
		limit = { check_variable = { dd_cw_militia_count > 0 } }
		capital_scope = {
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.6"
				owner = PREV
				count = 1
			}
		}
		subtract_from_variable = { dd_cw_militia_count = 1 }
	}
}

# Ali Khamenei keeps the theocrat role after retire_country_leader. set_politics and
# start_civil_war put that history leader back in the chair.
dd_per_strip_khamenei = {
	if = {
		limit = { has_character = PER_ali_khamenei }
		PER_ali_khamenei = { remove_country_leader_role = { ideology = theocrat } }
	}
	every_character = {
		limit = {
			has_ideology = theocrat
			NOT = { is_character = PER_pakpour_head }
			NOT = { is_character = PER_masoud_pezeshkian }
			NOT = { is_character = PER_mojtaba_khamenei }
		}
		remove_country_leader_role = { ideology = theocrat }
	}
}
