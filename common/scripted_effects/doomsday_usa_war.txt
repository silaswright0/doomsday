# Country-scoped. Caller sets dd_cw_spawn. create_unit is state-scoped only.
# Spreads across owned states, then pocket provinces, then capital (MP-safe, no random).
dd_usa_cw_spawn_militia_n = {
	dd_usa_cw_tpl_militia_2x6 = yes
	set_temp_variable = { dd_cw_eq = dd_cw_spawn }
	multiply_temp_variable = { dd_cw_eq = 1500 }
	add_equipment_to_stockpile = { type = infantry_equipment_cw1 amount = dd_cw_eq }
	# Reserve ~1/3 of the force for pocket provinces when any exist.
	set_variable = { dd_cw_pocket_budget = dd_cw_spawn }
	multiply_variable = { dd_cw_pocket_budget = 0.34 }
	round_variable = dd_cw_pocket_budget
	subtract_from_variable = { dd_cw_spawn = dd_cw_pocket_budget }
	while_loop_effect = {
		limit = { check_variable = { dd_cw_spawn > 0 } }
		set_temp_variable = { dd_cw_pass = 0 }
		every_owned_state = {
			limit = {
				is_owned_and_controlled_by = PREV
				check_variable = { PREV.dd_cw_spawn > 0 }
			}
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
				owner = PREV
				count = 1
			}
			PREV = { subtract_from_variable = { dd_cw_spawn = 1 } }
			add_to_temp_variable = { dd_cw_pass = 1 }
		}
		every_controlled_state = {
			limit = {
				NOT = { is_owned_by = PREV }
				is_controlled_by = PREV
				check_variable = { PREV.dd_cw_spawn > 0 }
			}
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
				owner = PREV
				count = 1
				allow_spawning_on_enemy_provs = yes
			}
			PREV = { subtract_from_variable = { dd_cw_spawn = 1 } }
			add_to_temp_variable = { dd_cw_pass = 1 }
		}
		if = {
			limit = { check_variable = { dd_cw_pass < 1 } }
			capital_scope = {
				create_unit = {
					division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
					owner = PREV
					count = 1
				}
			}
			subtract_from_variable = { dd_cw_spawn = 1 }
		}
	}
	# Pocket pass
	set_variable = { dd_cw_spawn = dd_cw_pocket_budget }
	clear_variable = dd_cw_pocket_budget
	dd_usa_cw_spawn_militia_pockets = yes
	# Any unused pocket budget dumps to capital.
	if = {
		limit = { check_variable = { dd_cw_spawn > 0 } }
		capital_scope = {
			create_unit = {
				division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
				owner = PREV
				count = dd_cw_spawn
			}
		}
	}
	clear_variable = dd_cw_spawn
}

			PREV = { subtract_from_variable = { dd_cw_spawn = 1 } }
			add_to_temp_variable = { dd_cw_pass = 1 }
		}
		if = {

			capital_scope = {
				create_unit = {
					division = "name = \"Militia\" division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 0.55"
					owner = PREV

		}
