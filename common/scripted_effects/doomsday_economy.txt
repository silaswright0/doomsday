dd_apply_rates = {
	# GDP points, then × dd_prod, match real 2026 GDP at the start.
	# A new civilian factory adds more GDP than a military factory.
	set_variable = { dd_w_pop = 1 }
	set_variable = { dd_w_civ = 8 }
	set_variable = { dd_w_mil = 4 }
	set_variable = { dd_w_dock = 6 }
	# A financial centre is local. Base, plus civilian factories in that state, plus services there.
	# Caps stop one megacity from minting points. Services stay a flat 5 and are cheap to build.
	set_variable = { dd_fin_base = 6 }
	set_variable = { dd_fin_civ_cap = 8 }
	set_variable = { dd_fin_srv_cap = 4 }
	set_variable = { dd_w_srv = 5 }
	# Tax law is a share of yearly GDP. No Taxes is sales and tax on foreign firms, not a wage tax.
	set_variable = { dd_tax_share_1 = 0.03 }
	set_variable = { dd_tax_share_2 = 0.18 }
	set_variable = { dd_tax_share_3 = 0.22 }
	set_variable = { dd_tax_share_4 = 0.28 }
	set_variable = { dd_tax_share_5 = 0.34 }
	# Oil spirit: billions per controlled oil unit per month. Checkpoint spirit is a flat monthly sum.
	set_variable = { dd_oil_rent = 0.05 }
	set_variable = { dd_checkpoint_income = 0.02 }
	# Core state, before policy tiers. Policy adds more shares of GDP.
	set_variable = { dd_spend_base = 0.05 }
	# 3% a year on the debt stock (US Treasury average coupon is ~3.3% in 2026).
	set_variable = { dd_rate_interest = 0.03 }
	divide_variable = { dd_rate_interest = 12 }
	# Japan home-bias spirit: −2 percentage points a year.
	set_variable = { dd_rate_interest_jap_cut = 0.02 }
	# Decision prices are a share of yearly revenue.
	set_variable = { dd_rate_sub_cost = 0.005 }
	set_variable = { dd_rate_pp_cost = 0.01 }
}

dd_refresh_interest = {
	set_variable = { expense_interest = debt }
	multiply_variable = { expense_interest = dd_rate_interest }
	if = {
		limit = { has_idea = JAP_boj_home_bias }
		set_temp_variable = { jap_cut = debt }
		multiply_temp_variable = { jap_cut = dd_rate_interest_jap_cut }
		divide_temp_variable = { jap_cut = 12 }
		subtract_from_variable = { expense_interest = jap_cut }
	}
	if = {
		limit = { check_variable = { expense_interest < 0 } }
		set_variable = { expense_interest = 0 }
	}
	# Compact debt UI shows the yearly coupon, not the stock.
	set_variable = { dd_debt_year = expense_interest }
	multiply_variable = { dd_debt_year = 12 }
}

dd_refresh_loan_preview = {
	set_variable = { dd_loan_small = dd_last_income }
	set_variable = { dd_loan_large = dd_last_income }
	multiply_variable = { dd_loan_large = 5 }
	set_variable = { dd_repay_preview = dd_loan_small }
	if = {
		limit = { check_variable = { treasury < dd_repay_preview } }
		set_variable = { dd_repay_preview = treasury }
	}
	if = {
		limit = { check_variable = { debt < dd_repay_preview } }
		set_variable = { dd_repay_preview = debt }
	}
	dd_refresh_usd_display = yes
}

# Decision prices scale with yearly revenue. Subsidy 0.5%, PP buy 1%.
dd_refresh_decision_costs = {
	set_variable = { dd_year_rev = dd_last_income }
	multiply_variable = { dd_year_rev = 12 }
	set_variable = { dd_cost_subsidize = dd_year_rev }
	multiply_variable = { dd_cost_subsidize = dd_rate_sub_cost }
	set_variable = { dd_cost_pp = dd_year_rev }
	multiply_variable = { dd_cost_pp = dd_rate_pp_cost }
	set_variable = { dd_cost_sub_m = dd_cost_subsidize }
	multiply_variable = { dd_cost_sub_m = 1000 }
	set_variable = { dd_cost_pp_m = dd_cost_pp }
	multiply_variable = { dd_cost_pp_m = 1000 }
}

# Debt / twelve months of revenue. One stack per full 100% (floor).
# 50% → 0, 150% → 1, 280% → 2, 305% → 3. Each stack: −10% construction, −2% stability.
dd_refresh_debt_burden = {
	if = {
		limit = { NOT = { has_country_flag = dd_econ_ledger_ready } }
		set_variable = { dd_debt_rev_ratio = 0 }
		set_variable = { dd_debt_burden_stacks = 0 }
	}
	else_if = {
		limit = { check_variable = { dd_last_income < 0.01 } }
		if = {
			limit = { check_variable = { debt > 0.01 } }
			set_variable = { dd_debt_rev_ratio = 9999 }
			set_variable = { dd_debt_burden_stacks = 50 }
		}
		else = {
			set_variable = { dd_debt_rev_ratio = 0 }
			set_variable = { dd_debt_burden_stacks = 0 }
		}
	}
	else = {
		set_variable = { dd_year_rev = dd_last_income }
		multiply_variable = { dd_year_rev = 12 }
		set_variable = { dd_debt_rev_ratio = debt }
		divide_variable = { dd_debt_rev_ratio = dd_year_rev }
		multiply_variable = { dd_debt_rev_ratio = 100 }
		set_variable = { dd_debt_burden_stacks = debt }
		divide_variable = { dd_debt_burden_stacks = dd_year_rev }
		subtract_from_variable = { dd_debt_burden_stacks = 0.499 }
		round_variable = dd_debt_burden_stacks
		if = {
			limit = { check_variable = { dd_debt_burden_stacks < 0 } }
			set_variable = { dd_debt_burden_stacks = 0 }
		}
	}
	set_variable = { dd_debt_con_pen = dd_debt_burden_stacks }
	multiply_variable = { dd_debt_con_pen = -0.1 }
	set_variable = { dd_debt_stab_pen = dd_debt_burden_stacks }
	multiply_variable = { dd_debt_stab_pen = -0.02 }
	if = {
		limit = { check_variable = { dd_debt_burden_stacks > 0.9 } }
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = dd_debt_burden } }
			}
			add_dynamic_modifier = { modifier = dd_debt_burden }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_debt_burden } }
			remove_dynamic_modifier = { modifier = dd_debt_burden }
		}
	}
}

dd_take_loan_small = {
	if = {
		limit = {
			check_variable = { dd_loan_small > 0.01 }
		}
		add_to_variable = { treasury = dd_loan_small }
		add_to_variable = { debt = dd_loan_small }
		dd_refresh_interest = yes
		dd_refresh_debt_burden = yes
		dd_refresh_loan_preview = yes
	}
}

dd_take_loan_large = {
	if = {
		limit = {
			check_variable = { dd_loan_large > 0.01 }
		}
		add_to_variable = { treasury = dd_loan_large }
		add_to_variable = { debt = dd_loan_large }
		dd_refresh_interest = yes
		dd_refresh_debt_burden = yes
		dd_refresh_loan_preview = yes
	}
}

dd_repay_loan = {
	dd_refresh_loan_preview = yes
	if = {
		limit = { check_variable = { dd_repay_preview > 0.01 } }
		subtract_from_variable = { treasury = dd_repay_preview }
		subtract_from_variable = { debt = dd_repay_preview }
		dd_refresh_interest = yes
		dd_refresh_debt_burden = yes
		dd_refresh_loan_preview = yes
	}
}

# Cover an overdraft automatically. One loan: at least 1× monthly revenue,
# or exactly enough to reach 0 if the hole is larger. Floor 0.1B if income is 0.
dd_auto_loan_if_negative = {
	if = {
		limit = { check_variable = { treasury < 0 } }
		dd_refresh_loan_preview = yes
		if = {
			limit = { NOT = { check_variable = { dd_loan_small > 0.1 } } }
			set_variable = { dd_loan_small = 0.1 }
		}
		set_temp_variable = { dd_cover = treasury }
		multiply_temp_variable = { dd_cover = -1 }
		if = {
			limit = { check_variable = { dd_cover < dd_loan_small } }
			set_temp_variable = { dd_cover = dd_loan_small }
		}
		add_to_variable = { treasury = dd_cover }
		add_to_variable = { debt = dd_cover }
		dd_refresh_interest = yes
		dd_refresh_debt_burden = yes
		dd_refresh_loan_preview = yes
	}
}

# Pay as much debt as cash allows (leaves leftover cash if debt is smaller).
dd_auto_repay_debt = {
	if = {
		limit = {
			has_country_flag = dd_auto_repay
			check_variable = { debt > 0.01 }
			check_variable = { treasury > 0.01 }
		}
		set_temp_variable = { dd_auto_pay = treasury }
		if = {
			limit = { check_variable = { debt < dd_auto_pay } }
			set_temp_variable = { dd_auto_pay = debt }
		}
		subtract_from_variable = { treasury = dd_auto_pay }
		subtract_from_variable = { debt = dd_auto_pay }
		dd_refresh_interest = yes
		dd_refresh_debt_burden = yes

dd_recalc_accounts = {
	dd_apply_rates = yes
	if = {
		limit = { check_variable = { dd_prod < 0.0001 } }
		set_variable = { dd_prod = 1 }
	}
	set_variable = { dd_pop_m = 0 }
	set_variable = { dd_n_civ = 0 }
	set_variable = { dd_n_mil = 0 }
	set_variable = { dd_n_dock = 0 }
	set_variable = { dd_n_fin = 0 }
	set_variable = { dd_n_srv = 0 }
	set_variable = { dd_pts_fin = 0 }
	every_controlled_state = {
		set_temp_variable = { pop_m = state_population_k }
		divide_temp_variable = { pop_m = 1000 }
		multiply_temp_variable = { pop_m = 10 }
		round_temp_variable = pop_m
		divide_temp_variable = { pop_m = 10 }
		add_to_variable = { ROOT.dd_pop_m = pop_m }
		set_temp_variable = { st_civ = building_level@industrial_complex }
		set_temp_variable = { st_mil = building_level@arms_factory }
		set_temp_variable = { st_dock = building_level@dockyard }
		set_temp_variable = { st_fin = building_level@finance_center }
		set_temp_variable = { st_srv = building_level@services_building }
		add_to_variable = { ROOT.dd_n_civ = st_civ }
		add_to_variable = { ROOT.dd_n_mil = st_mil }
		add_to_variable = { ROOT.dd_n_dock = st_dock }
		add_to_variable = { ROOT.dd_n_fin = st_fin }
		add_to_variable = { ROOT.dd_n_srv = st_srv }
		if = {
			limit = { check_variable = { st_fin > 0 } }
			set_temp_variable = { fin_civ = st_civ }
			if = {
				limit = { check_variable = { fin_civ > ROOT.dd_fin_civ_cap } }
				set_temp_variable = { fin_civ = ROOT.dd_fin_civ_cap }
			}
			set_temp_variable = { fin_srv = st_srv }
			if = {
				limit = { check_variable = { fin_srv > ROOT.dd_fin_srv_cap } }
				set_temp_variable = { fin_srv = ROOT.dd_fin_srv_cap }
			}
			set_temp_variable = { fin_each = ROOT.dd_fin_base }
			add_to_temp_variable = { fin_each = fin_civ }
			add_to_temp_variable = { fin_each = fin_srv }
			multiply_temp_variable = { fin_each = st_fin }
			add_to_variable = { ROOT.dd_pts_fin = fin_each }
		}
	}
	set_variable = { dd_pts_pop = dd_pop_m }
	multiply_variable = { dd_pts_pop = dd_w_pop }
	set_variable = { dd_pts_civ = dd_n_civ }
	multiply_variable = { dd_pts_civ = dd_w_civ }
	set_variable = { dd_pts_mil = dd_n_mil }
	multiply_variable = { dd_pts_mil = dd_w_mil }
	set_variable = { dd_pts_dock = dd_n_dock }
	multiply_variable = { dd_pts_dock = dd_w_dock }
	set_variable = { dd_pts_srv = dd_n_srv }
	multiply_variable = { dd_pts_srv = dd_w_srv }
	set_variable = { dd_gdp_points = dd_pts_pop }
	add_to_variable = { dd_gdp_points = dd_pts_civ }
	add_to_variable = { dd_gdp_points = dd_pts_mil }
	add_to_variable = { dd_gdp_points = dd_pts_dock }
	add_to_variable = { dd_gdp_points = dd_pts_fin }
	add_to_variable = { dd_gdp_points = dd_pts_srv }
	set_variable = { dd_gdp = dd_gdp_points }
	multiply_variable = { dd_gdp = dd_prod }

	dd_recalc_policy_expense = yes
	set_variable = { dd_last_income = dd_gdp }
	multiply_variable = { dd_last_income = dd_tax_share }
	divide_variable = { dd_last_income = 12 }
	set_variable = { income_tax = 0 }
	set_variable = { income_civ = 0 }
	set_variable = { income_finance = 0 }
	set_variable = { income_services = 0 }
	if = {
		limit = { check_variable = { dd_gdp_points > 0.001 } }
		set_variable = { income_tax = dd_last_income }
		multiply_variable = { income_tax = dd_pts_pop }
		divide_variable = { income_tax = dd_gdp_points }
		set_variable = { income_civ = dd_pts_civ }
		add_to_variable = { income_civ = dd_pts_mil }
		add_to_variable = { income_civ = dd_pts_dock }
		multiply_variable = { income_civ = dd_last_income }
		divide_variable = { income_civ = dd_gdp_points }
		set_variable = { income_finance = dd_last_income }
		multiply_variable = { income_finance = dd_pts_fin }
		divide_variable = { income_finance = dd_gdp_points }
		set_variable = { income_services = dd_last_income }
		multiply_variable = { income_services = dd_pts_srv }
		divide_variable = { income_services = dd_gdp_points }
	}

	# Spirits are handed out in history. They are revenue, so debt uses them.
	set_variable = { dd_oil_units = 0 }
	set_variable = { dd_income_oil = 0 }
	set_variable = { dd_income_checkpoint = 0 }
	if = {
		limit = { has_idea = dd_oil_revenue }
		every_controlled_state = {
			set_temp_variable = { st_oil = resource@oil }
			add_to_variable = { ROOT.dd_oil_units = st_oil }
		}
		set_variable = { dd_income_oil = dd_oil_units }
		multiply_variable = { dd_income_oil = dd_oil_rent }
	}
	if = {
		limit = { has_idea = dd_checkpoint_revenue }
		set_variable = { dd_income_checkpoint = dd_checkpoint_income }
	}
	add_to_variable = { dd_last_income = dd_income_oil }
	add_to_variable = { dd_last_income = dd_income_checkpoint }

	set_variable = { expense_admin = dd_gdp }
	multiply_variable = { expense_admin = dd_spend_base }
	divide_variable = { expense_admin = 12 }
	set_variable = { expense_policy = dd_gdp }
	multiply_variable = { expense_policy = dd_spend_share }
	divide_variable = { expense_policy = 12 }
	subtract_from_variable = { expense_policy = expense_admin }
	dd_refresh_interest = yes

	set_variable = { dd_last_expense = expense_admin }
	add_to_variable = { dd_last_expense = expense_policy }
	add_to_variable = { dd_last_expense = expense_interest }

	set_variable = { dd_last_net = dd_last_income }
	subtract_from_variable = { dd_last_net = dd_last_expense }

	dd_refresh_debt_burden = yes
	dd_refresh_decision_costs = yes
	dd_refresh_loan_preview = yes
	dd_refresh_market_counts = yes
}

dd_economy_monthly_tick = {
	dd_init_economy = yes
	clr_country_flag = dd_pp_bought
	dd_recalc_accounts = yes
	# Do not book a month that was calculated before factories existed.
	if = {
		limit = { has_country_flag = dd_econ_ledger_ready }
		add_to_variable = { treasury = dd_last_income }
		subtract_from_variable = { treasury = dd_last_expense }
		# First posted close reveals the budget UI (January stays ??).
		set_country_flag = dd_econ_budget_shown
		dd_refresh_loan_preview = yes
		# Overdraft → small loans first, then optional auto debt paydown.
		dd_auto_loan_if_negative = yes
		dd_auto_repay_debt = yes
		dd_consume_sam_ammo = yes
	}

	if = {
		limit = { has_global_flag = dd_econ_log }
		log = "DD_ECON [GetDateText] [ROOT.GetTag] treasury=[?treasury] debt=[?debt] income=[?dd_last_income] expense=[?dd_last_expense]"
	}
}

dd_consume_sam_ammo = {
	# Airbase SAM interceptors spend sam_missile_equipment via one_use_only.
	# Only mobile battalions need a monthly stockpile tick.
	set_variable = { dd_sam_need = num_battalions_with_type@mobile_sam }
	if = {
		limit = { check_variable = { dd_sam_need > 0 } }
		for_loop_effect = {
			start = 0
			end = dd_sam_need
			compare = less_than
			add = 1
			value = v
			if = {
				limit = { has_equipment = { sam_missile_equipment > 0 } }
				add_equipment_to_stockpile = {
					type = sam_missile_equipment
					amount = -1
					producer = ROOT
				}
			}
		}
	}
}
