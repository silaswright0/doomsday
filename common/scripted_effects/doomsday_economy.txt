dd_apply_rates = {
	# Ledger is US$ billions / month. Fit to IMF WEO central-ish budgets after
	# the 2026 pop/building rebuild. Same tax law for every tag at start, so
	# rates target a standard federal take, not France-at-rest. Pop × building
	# income is not used: Indian mega-states would mint more cash than the USA.
	set_variable = { dd_rate_civ = 1.00 }
	# Finance: level × (local civs + services + 4). Services: level × (civs + 2).
	set_variable = { dd_rate_finance = 0.20 }
	set_variable = { dd_rate_fin_floor = 4 }
	set_variable = { dd_rate_services = 0.15 }
	set_variable = { dd_rate_srv_floor = 2 }
	# State cost: people plus factories in owned states.
	set_variable = { dd_rate_admin_pop = 0.02 }
	set_variable = { dd_rate_admin_civ = 1.20 }
	set_variable = { dd_rate_admin = 0.02 }
	# 3% a year on the debt stock (US Treasury average coupon is ~3.3% in 2026).
	set_variable = { dd_rate_interest = 0.03 }
	divide_variable = { dd_rate_interest = 12 }
	# Japan home-bias spirit: −2 percentage points a year.
	set_variable = { dd_rate_interest_jap_cut = 0.02 }
	# Standard tax (law 2). Law 3 doubles. Law 1 is zero.
	set_variable = { dd_rate_tax_pop = 0.02 }
	set_variable = { dd_rate_tax_civ = 0.80 }
	set_variable = { dd_rate_tax_2 = 0.02 }
	set_variable = { dd_rate_tax_3 = 2.00 }
	# Decision prices are a share of yearly revenue.
	set_variable = { dd_rate_sub_cost = 0.005 }
	set_variable = { dd_rate_pp_cost = 0.01 }
}

dd_refresh_interest = {
	set_variable = { expense_interest = debt }
	multiply_variable = { expense_interest = dd_rate_interest }
	if = {
		limit = { has_idea = JAP_boj_home_bias }
		set_temp_variable = { jap_cut = debt }
		multiply_temp_variable = { jap_cut = dd_rate_interest_jap_cut }
		divide_temp_variable = { jap_cut = 12 }
		subtract_from_variable = { expense_interest = jap_cut }
	}
	if = {
		limit = { check_variable = { expense_interest < 0 } }
		set_variable = { expense_interest = 0 }
	}
	# Compact debt UI shows the yearly coupon, not the stock.
	set_variable = { dd_debt_year = expense_interest }
	multiply_variable = { dd_debt_year = 12 }
}

dd_refresh_loan_preview = {
	set_variable = { dd_loan_small = dd_last_income }
	set_variable = { dd_loan_large = dd_last_income }
	multiply_variable = { dd_loan_large = 5 }
	set_variable = { dd_repay_preview = dd_loan_small }
	if = {
		limit = { check_variable = { treasury < dd_repay_preview } }
		set_variable = { dd_repay_preview = treasury }
	}
	if = {
		limit = { check_variable = { debt < dd_repay_preview } }
		set_variable = { dd_repay_preview = debt }
	}
	dd_refresh_usd_display = yes
}

# Decision prices scale with yearly revenue. Subsidy 0.5%, PP buy 1%.
dd_refresh_decision_costs = {
	set_variable = { dd_year_rev = dd_last_income }
	multiply_variable = { dd_year_rev = 12 }
	set_variable = { dd_cost_subsidize = dd_year_rev }
	multiply_variable = { dd_cost_subsidize = dd_rate_sub_cost }
	set_variable = { dd_cost_pp = dd_year_rev }
	multiply_variable = { dd_cost_pp = dd_rate_pp_cost }
	set_variable = { dd_cost_sub_m = dd_cost_subsidize }
	multiply_variable = { dd_cost_sub_m = 1000 }
	set_variable = { dd_cost_pp_m = dd_cost_pp }
	multiply_variable = { dd_cost_pp_m = 1000 }
}

# Debt / twelve months of revenue. One stack per full 100% (floor).
# 50% → 0, 150% → 1, 280% → 2, 305% → 3. Each stack: −10% construction, −2% stability.
dd_refresh_debt_burden = {
	if = {
		limit = { NOT = { has_country_flag = dd_econ_ledger_ready } }
		set_variable = { dd_debt_rev_ratio = 0 }
		set_variable = { dd_debt_burden_stacks = 0 }
	}
	else_if = {
		limit = { check_variable = { dd_last_income < 0.01 } }
		if = {
			limit = { check_variable = { debt > 0.01 } }
			set_variable = { dd_debt_rev_ratio = 9999 }
			set_variable = { dd_debt_burden_stacks = 50 }
		}
		else = {
			set_variable = { dd_debt_rev_ratio = 0 }
			set_variable = { dd_debt_burden_stacks = 0 }
		}
	}
	else = {
		set_variable = { dd_year_rev = dd_last_income }
		multiply_variable = { dd_year_rev = 12 }
		set_variable = { dd_debt_rev_ratio = debt }
		divide_variable = { dd_debt_rev_ratio = dd_year_rev }
		multiply_variable = { dd_debt_rev_ratio = 100 }
		set_variable = { dd_debt_burden_stacks = debt }
		divide_variable = { dd_debt_burden_stacks = dd_year_rev }
		subtract_from_variable = { dd_debt_burden_stacks = 0.499 }
		round_variable = dd_debt_burden_stacks
		if = {
			limit = { check_variable = { dd_debt_burden_stacks < 0 } }
			set_variable = { dd_debt_burden_stacks = 0 }
		}
	}
	set_variable = { dd_debt_con_pen = dd_debt_burden_stacks }
	multiply_variable = { dd_debt_con_pen = -0.1 }
	set_variable = { dd_debt_stab_pen = dd_debt_burden_stacks }
	multiply_variable = { dd_debt_stab_pen = -0.02 }
	if = {
		limit = { check_variable = { dd_debt_burden_stacks > 0.9 } }
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = dd_debt_burden } }
			}
			add_dynamic_modifier = { modifier = dd_debt_burden }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_debt_burden } }
			remove_dynamic_modifier = { modifier = dd_debt_burden }
		}
	}
}

	set_variable = { dd_admin_factor = 1.00 }

dd_recalc_accounts = {
	dd_apply_rates = yes
	set_variable = { income_civ = num_of_civilian_factories }
	multiply_variable = { income_civ = dd_rate_civ }

	set_variable = { income_finance = 0 }
	set_variable = { income_services = 0 }
	every_controlled_state = {
		set_temp_variable = { tmp_mkt = building_level@industrial_complex }
		add_to_temp_variable = { tmp_mkt = building_level@services_building }
		add_to_temp_variable = { tmp_mkt = ROOT.dd_rate_fin_floor }
		set_temp_variable = { tmp_fin = building_level@finance_center }
		multiply_temp_variable = { tmp_fin = tmp_mkt }
		multiply_temp_variable = { tmp_fin = ROOT.dd_rate_finance }
		add_to_variable = { ROOT.income_finance = tmp_fin }
		set_temp_variable = { tmp_mkt = building_level@industrial_complex }
		add_to_temp_variable = { tmp_mkt = ROOT.dd_rate_srv_floor }
		set_temp_variable = { tmp_srv = building_level@services_building }
		multiply_temp_variable = { tmp_srv = tmp_mkt }
		multiply_temp_variable = { tmp_srv = ROOT.dd_rate_services }
		add_to_variable = { ROOT.income_services = tmp_srv }
	}

	set_variable = { dd_pop_m = 0 }
	set_variable = { dd_tax_base = 0 }
	set_variable = { expense_admin = 0 }
	every_owned_state = {
		set_temp_variable = { pop_m = state_population_k }
		divide_temp_variable = { pop_m = 1000 }
		multiply_temp_variable = { pop_m = 10 }
		round_temp_variable = pop_m
		divide_temp_variable = { pop_m = 10 }
		add_to_variable = { ROOT.dd_pop_m = pop_m }
		set_temp_variable = { st_civ = building_level@industrial_complex }
		set_temp_variable = { st_admin = pop_m }
		multiply_temp_variable = { st_admin = ROOT.dd_rate_admin_pop }
		set_temp_variable = { st_adm_c = st_civ }
		multiply_temp_variable = { st_adm_c = ROOT.dd_rate_admin_civ }
		add_to_temp_variable = { st_admin = st_adm_c }
		add_to_variable = { ROOT.expense_admin = st_admin }
		set_temp_variable = { st_tax = pop_m }
		multiply_temp_variable = { st_tax = ROOT.dd_rate_tax_pop }
		set_temp_variable = { st_tax_c = st_civ }
		multiply_temp_variable = { st_tax_c = ROOT.dd_rate_tax_civ }
		add_to_temp_variable = { st_tax = st_tax_c }
		add_to_variable = { ROOT.dd_tax_base = st_tax }
	}

	dd_refresh_interest = yes
	dd_recalc_policy_expense = yes

	set_variable = { dd_last_income = income_civ }
	add_to_variable = { dd_last_income = income_finance }
	add_to_variable = { dd_last_income = income_services }
	add_to_variable = { dd_last_income = income_tax }

	set_variable = { dd_last_expense = expense_policy }
	add_to_variable = { dd_last_expense = expense_admin }
	add_to_variable = { dd_last_expense = expense_interest }

	set_variable = { dd_last_net = dd_last_income }
	subtract_from_variable = { dd_last_net = dd_last_expense }

	dd_refresh_debt_burden = yes
	dd_refresh_decision_costs = yes
	dd_refresh_loan_preview = yes
	dd_refresh_market_counts = yes
}

dd_economy_monthly_tick = {
	dd_init_economy = yes
	clr_country_flag = dd_pp_bought
	dd_recalc_accounts = yes
	# Do not book a month that was calculated before factories existed.
	if = {
		limit = { has_country_flag = dd_econ_ledger_ready }
		add_to_variable = { treasury = dd_last_income }
		subtract_from_variable = { treasury = dd_last_expense }
		# First posted close reveals the budget UI (January stays ??).
		set_country_flag = dd_econ_budget_shown
		dd_refresh_loan_preview = yes
		dd_consume_sam_ammo = yes
	}

	if = {
		limit = { has_global_flag = dd_econ_log }
		log = "DD_ECON [GetDateText] [ROOT.GetTag] treasury=[?treasury] debt=[?debt] income=[?dd_last_income] expense=[?dd_last_expense]"
	}
}

dd_consume_sam_ammo = {
	# Airbase SAM interceptors spend sam_missile_equipment via one_use_only.
	# Only mobile battalions need a monthly stockpile tick.
	set_variable = { dd_sam_need = num_battalions_with_type@mobile_sam }
	if = {
		limit = { check_variable = { dd_sam_need > 0 } }
		for_loop_effect = {
			start = 0
			end = dd_sam_need
			compare = less_than
			add = 1
			value = v
			if = {
				limit = { has_equipment = { sam_missile_equipment > 0 } }
				add_equipment_to_stockpile = {
					type = sam_missile_equipment
					amount = -1
					producer = ROOT
				}
			}
		}
	}
}
