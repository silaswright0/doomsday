# Deep refineries only. A silo on the contact line is the front, not a strategic strike.
dd_ukr_check_refinery_strike = {
	if = {
		limit = {
			has_country_flag = dd_ukr_diesel_watch
			has_war_with = SOV
		}
		every_state = {
			limit = {
				is_controlled_by = SOV
				NOT = { is_core_of = UKR }
				NOT = { any_neighbor_state = { is_controlled_by = UKR } }
				OR = {
					synthetic_refinery > 0
					fuel_silo > 0
				}
			}
			if = {
				limit = {
					OR = {
						check_variable = { damaged_building_level@fuel_silo > 0 }
						check_variable = { damaged_building_level@synthetic_refinery > 0 }
					}
				}
				if = {
					limit = {
						NOT = { has_state_flag = dd_ukr_ref_seen }
						ROOT = { has_country_flag = dd_ukr_ref_baseline }
					}
					set_state_flag = dd_ukr_ref_seen
					ROOT = { set_country_flag = dd_ukr_new_ref_hit }
				}
				else_if = {
					limit = { NOT = { has_state_flag = dd_ukr_ref_seen } }
					set_state_flag = dd_ukr_ref_seen
				}
			}
			else = {
				clr_state_flag = dd_ukr_ref_seen
			}
		}
		if = {
			limit = { has_country_flag = dd_ukr_new_ref_hit }
			clr_country_flag = dd_ukr_new_ref_hit
			if = {
				limit = { NOT = { has_idea = dd_ukr_starlink_disabled } }
				add_timed_idea = { idea = dd_ukr_starlink_disabled days = 7 }
				country_event = { id = dd_ukr_aid.4 }
			}
		}
		set_country_flag = dd_ukr_ref_baseline
	}
}

		limit = { is_controlled_by = SOV }

	}

# Ukraine scope. Writes the blackout penalties and shows or hides the modifier.
dd_ukr_grid_sync = {
	if = {
		limit = { NOT = { has_variable = dd_ukr_blackout_cap } }
		set_variable = { dd_ukr_blackout_cap = 20 }
	}
	if = {
		limit = { check_variable = { dd_ukr_blackout > dd_ukr_blackout_cap } }
		set_variable = { dd_ukr_blackout = dd_ukr_blackout_cap }
	}
	if = {
		limit = { check_variable = { dd_ukr_blackout < 0 } }
		set_variable = { dd_ukr_blackout = 0 }
	}
	set_variable = { dd_ukr_grid_fac = dd_ukr_blackout }
	multiply_variable = { dd_ukr_grid_fac = -0.04 }
	set_variable = { dd_ukr_grid_stab = dd_ukr_blackout }
	multiply_variable = { dd_ukr_grid_stab = -0.02 }
	set_variable = { dd_ukr_grid_con = dd_ukr_blackout }
	multiply_variable = { dd_ukr_grid_con = -0.03 }
	set_variable = { dd_ukr_grid_cg = dd_ukr_blackout }
	multiply_variable = { dd_ukr_grid_cg = 0.02 }
	if = {
		limit = { check_variable = { dd_ukr_blackout > 0 } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = dd_ukr_grid_blackout } } }
			add_dynamic_modifier = { modifier = dd_ukr_grid_blackout }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = dd_ukr_grid_blackout } }
			remove_dynamic_modifier = { modifier = dd_ukr_grid_blackout }
		}
	}
}

# Ukraine scope. Adds dd_ukr_hit_amount stacks, capped, and tells Ukraine.
dd_ukr_grid_apply_hit = {
	if = {
		limit = { NOT = { has_variable = dd_ukr_blackout_cap } }
		set_variable = { dd_ukr_blackout_cap = 20 }
	}
	if = {
		limit = { check_variable = { dd_ukr_blackout < dd_ukr_blackout_cap } }
		add_to_variable = { dd_ukr_blackout = dd_ukr_hit_amount }
		dd_ukr_grid_sync = yes
		country_event = { id = dd_ukr_grid.1 }
	}
}
