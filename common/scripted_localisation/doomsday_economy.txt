# Ledger store is US$ billions. Show millions when abs(value) < 1.
# Never add a T (trillions) key. Stay in M/B even past $1000B.
# Compact debt lines show yearly interest (3% a year, Japan 1%), not the stock.
# Budget flow (revenue / expenses / net) stays ?? until the first monthly close.

defined_text = {
	name = GetDdCash
	text = {
		trigger = {
			check_variable = { treasury < 1 }
			check_variable = { treasury > -1 }
		}
		localization_key = DD_USD_CASH_M
	}
	text = {
		localization_key = DD_USD_CASH_B
	}
}
