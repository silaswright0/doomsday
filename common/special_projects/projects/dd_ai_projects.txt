# Machine intelligence. Own electronics facility, not the air lab.
# The facility uses the existing special_project_facility_spawn pins.

sp_dd_machine_intelligence = {
	specialization = specialization_electronics
	icon = GFX_sp_nuclear_radar

	project_tags = sp_tag_ai

	prototype_time = sp_time.prototype.long
	complexity = sp_complexity.large

	breakthrough_cost = {
		specialization_electronics = 2
	}

	resource_cost = {
		resources = {
			steel = 4
			aluminium = 8
		}
	}

	ai_will_do = {
		base = 1
		modifier = {
			factor = 4
			ROOT = { is_major = yes }
		}
	}

	project_output = {
		country_effects = {
			if = {
				limit = { NOT = { has_idea = dd_ai_program } }
				add_ideas = dd_ai_program
			}
		}
	}

	generic_prototype_rewards = {
		sp_air_scientist_xp_generic_reward
		sp_nuclear_generic_electronics_tech_reward
	}
}
