# February 28, 2026. US and Israel strike Iran. The strait closes the same day.
country_event = {
	id = dd_hormuz.1
	hidden = yes
	is_triggered_only = yes
	trigger = {
		NOT = { has_global_flag = dd_hormuz_war }
		PER = { exists = yes }
		ISR = { exists = yes }
	}
	immediate = {
		set_global_flag = dd_hormuz_war
		PER = { set_variable = { dd_hormuz_auth = 2 } }
		set_variable = { global.dd_hormuz_auth = 0 }
		set_variable = { global.dd_hormuz_quarter_elapsed = 0 }
		set_variable = { global.dd_hormuz_quarter_bomb_days = 0 }
		if = {
			limit = { NOT = { has_war_with = PER } }
			declare_war_on = { target = PER type = annex_everything }
		}
		if = {
			limit = { NOT = { ISR = { has_war_with = PER } } }
			ISR = {
				add_to_war = {
					targeted_alliance = USA
					enemy = PER
				}
			}
		}
		dd_hormuz_sync = yes
		add_ideas = dd_usa_iran_war
		dd_hormuz_join_gulf = yes
		PER = {
			dd_per_puppet_proxies = yes
			country_event = { id = dd_iran.1 }
		}
	}
	option = { name = dd_hormuz.1.a }
}
