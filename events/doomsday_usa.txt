	immediate = {
		set_variable = { dd_nato_failed = 0 }

			limit = {
				has_country_flag = dd_nato_ultimatum

		}
		set_temp_variable = { dd_nato_half = dd_nato_total }
		multiply_temp_variable = { dd_nato_half = 0.5 }

			limit = { check_variable = { dd_nato_failed > dd_nato_half } }

			news_event = { id = dd_usa_news.1 }
		}
	}
	option = {
		name = dd_usa_nato.1.a

	immediate = {

			if = {
				limit = { is_in_faction = no }

			}
			ETX = { dd_cw_restore_schism = yes }
		}

	}
}
